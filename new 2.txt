8. Auth Store
Şimdi authentication state’imizi Angular Signals ile yöneteceğiz.
Dosya:
src/app/core/auth/services/auth-store.service.ts

import { computed, Injectable, signal } from '@angular/core';

import { AuthSessionResponse } from '../models/auth-session-response.model';
import { AuthState } from '../models/auth-state.model';
import { AuthUser } from '../models/auth-user.model';

/**
 * Angular uygulamasındaki authentication durumunu
 * bellek içerisinde yöneten merkezi state servisidir.
 *
 * Access token yalnızca memory içerisinde tutulur.
 * Refresh token bu servisin erişimine hiçbir zaman verilmez.
 */
@Injectable({
  providedIn: 'root'
})
export class AuthStoreService {

  /**
   * Authentication state'inin değiştirilebilir dahili Signal nesnesidir.
   */
  private readonly _state = signal<AuthState>({
    isInitialized: false,
    isAuthenticated: false,
    accessToken: null,
    accessTokenExpiresAt: null,
    user: null
  });

  /**
   * Authentication state'inin dışarıya salt okunur olarak
   * sunulan halidir.
   */
  public readonly state = this._state.asReadonly();

  /**
   * Session initialization işleminin tamamlanıp
   * tamamlanmadığını belirtir.
   */
  public readonly isInitialized = computed(
    () => this._state().isInitialized
  );

  /**
   * Kullanıcının oturum açmış olup olmadığını belirtir.
   */
  public readonly isAuthenticated = computed(
    () => this._state().isAuthenticated
  );

  /**
   * Aktif access token bilgisini döndürür.
   */
  public readonly accessToken = computed(
    () => this._state().accessToken
  );

  /**
   * Güncel kullanıcı bilgisini döndürür.
   */
  public readonly user = computed(
    () => this._state().user
  );

  /**
   * Backend'den alınan başarılı login veya refresh sonucunu
   * authentication state'ine yazar.
   *
   * @param session Backend tarafından dönen authentication session bilgisidir.
   */
  public setSession(session: AuthSessionResponse): void {
    this._state.set({
      isInitialized: true,
      isAuthenticated: true,
      accessToken: session.accessToken,
      accessTokenExpiresAt: session.accessTokenExpiresAt,
      user: session.user
    });
  }

  /**
   * Mevcut kullanıcı bilgisini günceller.
   *
   * Access token ve diğer authentication bilgileri korunur.
   *
   * @param user Güncel kullanıcı bilgisidir.
   */
  public setUser(user: AuthUser): void {
    this._state.update(current => ({
      ...current,
      user
    }));
  }

  /**
   * Authentication state'ini temizler.
   *
   * Bu işlem refresh token'a erişmez veya cookie silmez.
   * Refresh token backend tarafından yönetilir.
   */
  public clearSession(): void {
    this._state.set({
      isInitialized: true,
      isAuthenticated: false,
      accessToken: null,
      accessTokenExpiresAt: null,
      user: null
    });
  }

  /**
   * Mevcut access token değerini döndürür.
   *
   * HTTP interceptor tarafından Authorization header oluşturmak
   * amacıyla kullanılır.
   *
   * @returns JWT access token veya null.
   */
  public getAccessToken(): string | null {
    return this._state().accessToken;
  }
}

Dosya:
src/app/core/auth/services/auth-debug.service.ts

import { isDevMode, Injectable } from '@angular/core';

import { AuthSessionResponse } from '../models/auth-session-response.model';

/**
 * Authentication işlemleri sırasında development ortamında
 * response ve JWT bilgilerini tarayıcı console'una yazdıran servistir.
 *
 * Production ortamında çıktı üretmez.
 */
@Injectable({
  providedIn: 'root'
})
export class AuthDebugService {

  /**
   * Authentication response'unu ve çözümlenmiş JWT payload'ını
   * development console'una yazar.
   *
   * @param source Log kaynağını belirten açıklamadır.
   * @param response Backend authentication response'udur.
   */
  public logSession(
    source: string,
    response: AuthSessionResponse
  ): void {

    if (!isDevMode()) {
      return;
    }

    console.group(`[MSKBS AUTH] ${source}`);

    console.log('Authentication Response:', response);

    console.log(
      'Access Token:',
      response.accessToken
    );

    console.log(
      'Access Token Expires At:',
      response.accessTokenExpiresAt
    );

    console.log(
      'User:',
      response.user
    );

    console.log(
      'Decoded JWT Payload:',
      this.decodeJwtPayload(response.accessToken)
    );

    console.groupEnd();
  }

  /**
   * JWT access token'ın payload bölümünü yalnızca debug amacıyla çözer.
   *
   * Bu işlem JWT doğrulaması yapmaz.
   * Token doğrulaması her zaman backend tarafından yapılmalıdır.
   *
   * @param token Decode edilecek JWT token'dır.
   * @returns JWT payload nesnesi veya null.
   */
  public decodeJwtPayload(
    token: string
  ): Record<string, unknown> | null {

    try {
      const parts = token.split('.');

      if (parts.length !== 3) {
        return null;
      }

      const payload = parts[1]
        .replace(/-/g, '+')
        .replace(/_/g, '/');

      const paddedPayload =
        payload.padEnd(
          payload.length + ((4 - payload.length % 4) % 4),
          '='
        );

      const binary = atob(paddedPayload);

      const bytes = Uint8Array.from(
        binary,
        character => character.charCodeAt(0)
      );

      const json = new TextDecoder().decode(bytes);

      return JSON.parse(json) as Record<string, unknown>;
    }
    catch (error) {
      console.error(
        '[MSKBS AUTH] JWT decode edilemedi.',
        error
      );

      return null;
    }
  }
}

10. Ana AuthService
Şimdi en önemli servis.
Dosya:
src/app/core/auth/services/auth.service.ts

import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';

import {
  catchError,
  finalize,
  map,
  Observable,
  of,
  shareReplay,
  tap
} from 'rxjs';

import { API_BASE_URL } from '../../config/api.tokens';

import { LoginRequest } from '../models/login-request.model';
import { AuthSessionResponse } from '../models/auth-session-response.model';
import { AuthUser } from '../models/auth-user.model';

import { AuthStoreService } from './auth-store.service';
import { AuthDebugService } from './auth-debug.service';

/**
 * MSKBS authentication işlemlerinin Angular tarafındaki
 * merkezi servisidir.
 *
 * Login, refresh, session restore, current user ve logout
 * işlemlerinden sorumludur.
 */
@Injectable({
  providedIn: 'root'
})
export class AuthService {

  private readonly http = inject(HttpClient);

  private readonly apiBaseUrl = inject(API_BASE_URL);

  private readonly store = inject(AuthStoreService);

  private readonly debugService = inject(AuthDebugService);

  /**
   * Aynı anda birden fazla request 401 döndüğünde yalnızca
   * tek bir refresh isteğinin backend'e gönderilmesini sağlar.
   */
  private refreshInFlight$: Observable<AuthSessionResponse> | null = null;

  /**
   * Kullanıcının authentication durumunu dışarıya sunar.
   */
  public readonly isAuthenticated = this.store.isAuthenticated;

  /**
   * Session initialization durumunu dışarıya sunar.
   */
  public readonly isInitialized = this.store.isInitialized;

  /**
   * Oturum açmış güncel kullanıcı bilgisini dışarıya sunar.
   */
  public readonly currentUser = this.store.user;

  /**
   * Kullanıcı adı/e-posta ve parola ile oturum açar.
   *
   * Başarılı login sonucunda access token memory içerisine alınır.
   * Refresh token ise HttpOnly cookie olarak backend tarafından yönetilir.
   *
   * @param request Login bilgilerini içeren request modelidir.
   * @returns Authentication session response.
   */
  public login(
    request: LoginRequest
  ): Observable<AuthSessionResponse> {

    return this.http.post<AuthSessionResponse>(
      `${this.apiBaseUrl}/auth/login`,
      request,
      {
        withCredentials: true
      }
    )
    .pipe(
      tap(response => {

        this.store.setSession(response);

        this.debugService.logSession(
          'Login successful',
          response
        );
      })
    );
  }

  /**
   * HttpOnly cookie içerisindeki refresh token kullanılarak
   * yeni access token oluşturur.
   *
   * Eşzamanlı refresh talepleri tek bir HTTP isteği altında paylaşılır.
   *
   * @returns Yenilenmiş authentication session response.
   */
  public refresh(): Observable<AuthSessionResponse> {

    if (this.refreshInFlight$) {
      return this.refreshInFlight$;
    }

    const request$ = this.http.post<AuthSessionResponse>(
      `${this.apiBaseUrl}/auth/refresh`,
      {},
      {
        withCredentials: true
      }
    )
    .pipe(
      tap(response => {

        this.store.setSession(response);

        this.debugService.logSession(
          'Token refreshed',
          response
        );
      }),

      finalize(() => {
        this.refreshInFlight$ = null;
      }),

      shareReplay({
        bufferSize: 1,
        refCount: false
      })
    );

    this.refreshInFlight$ = request$;

    return request$;
  }

  /**
   * Angular uygulaması ilk açıldığında mevcut refresh cookie
   * üzerinden session'ı geri yüklemeye çalışır.
   *
   * Refresh cookie bulunmuyorsa kullanıcı anonymous olarak kabul edilir.
   *
   * @returns Initialization tamamlandığında sonuçlanan Observable.
   */
  public initializeSession(): Observable<void> {

    if (this.store.isInitialized()) {
      return of(void 0);
    }

    return this.refresh()
      .pipe(
        map(() => void 0),

        catchError(() => {

          this.store.clearSession();

          return of(void 0);
        })
      );
  }

  /**
   * Backend'den güncel kullanıcı bilgilerini getirir.
   *
   * @returns Güncel authenticated kullanıcı bilgisi.
   */
  public me(): Observable<AuthUser> {

    return this.http.get<AuthUser>(
      `${this.apiBaseUrl}/auth/me`,
      {
        withCredentials: true
      }
    )
    .pipe(
      tap(user => {

        this.store.setUser(user);

        console.log(
          '[MSKBS AUTH] /auth/me response:',
          user
        );
      })
    );
  }

  /**
   * Mevcut cihazdaki refresh token'ı revoke ederek
   * kullanıcının oturumunu kapatır.
   *
   * @returns Logout HTTP işlemi.
   */
  public logout(): Observable<void> {

    return this.http.post<void>(
      `${this.apiBaseUrl}/auth/logout`,
      {},
      {
        withCredentials: true
      }
    )
    .pipe(
      finalize(() => {
        this.store.clearSession();
      })
    );
  }

  /**
   * Kullanıcının bütün refresh token'larını revoke ederek
   * tüm cihazlardaki oturumlarını sonlandırır.
   *
   * @returns Logout-all HTTP işlemi.
   */
  public logoutAll(): Observable<void> {

    return this.http.post<void>(
      `${this.apiBaseUrl}/auth/logout-all`,
      {},
      {
        withCredentials: true
      }
    )
    .pipe(
      finalize(() => {
        this.store.clearSession();
      })
    );
  }

  /**
   * Angular memory içerisindeki access token değerini döndürür.
   *
   * HTTP interceptor tarafından kullanılır.
   *
   * @returns Aktif access token veya null.
   */
  public getAccessToken(): string | null {
    return this.store.getAccessToken();
  }

  /**
   * Angular tarafındaki authentication state'ini temizler.
   *
   * Genellikle refresh token doğrulaması başarısız olduğunda kullanılır.
   */
  public clearSession(): void {
    this.store.clearSession();
  }

  /**
   * Mevcut authentication state'ini development console'una yazar.
   */
  public debugCurrentSession(): void {

    console.log(
      '[MSKBS AUTH] Current Auth State:',
      this.store.state()
    );
  }
}

11. HTTP Interceptor
Dosya:
src/app/core/auth/interceptors/auth.interceptor.ts

import { inject } from '@angular/core';

import {
  HttpContextToken,
  HttpErrorResponse,
  HttpInterceptorFn,
  HttpRequest
} from '@angular/common/http';

import {
  catchError,
  switchMap,
  throwError
} from 'rxjs';

import { Router } from '@angular/router';

import { API_BASE_URL } from '../../config/api.tokens';
import { AuthService } from '../services/auth.service';

/**
 * Bir HTTP request'in refresh sonrasında tekrar gönderilip
 * gönderilmediğini belirtir.
 *
 * Sonsuz 401 → refresh → retry döngüsünü engeller.
 */
const AUTH_RETRY =
  new HttpContextToken<boolean>(() => false);

/**
 * MSKBS API çağrılarına access token ekleyen ve 401 durumlarında
 * otomatik refresh işlemini yöneten functional HTTP interceptor'dır.
 */
export const authInterceptor: HttpInterceptorFn =
  (request, next) => {

    const authService = inject(AuthService);

    const router = inject(Router);

    const apiBaseUrl = inject(API_BASE_URL);

    const isApiRequest =
      request.url.startsWith(apiBaseUrl);

    if (!isApiRequest) {
      return next(request);
    }

    const accessToken =
      authService.getAccessToken();

    let authenticatedRequest =
      addAuthentication(
        request,
        accessToken
      );

    return next(authenticatedRequest)
      .pipe(
        catchError((error: HttpErrorResponse) => {

          if (
            error.status !== 401 ||
            shouldSkipAutomaticRefresh(request.url) ||
            request.context.get(AUTH_RETRY)
          ) {

            return throwError(() => error);
          }

          /*
           * Request eski access token ile gönderilmiş olabilir.
           * Bu sırada başka bir request refresh işlemini tamamladıysa
           * ikinci kez refresh yapmak yerine yeni token ile tekrar deneriz.
           */
          const currentAccessToken =
            authService.getAccessToken();

          if (
            currentAccessToken &&
            accessToken &&
            currentAccessToken !== accessToken
          ) {

            authenticatedRequest =
              addAuthentication(
                request.clone({
                  context: request.context.set(
                    AUTH_RETRY,
                    true
                  )
                }),
                currentAccessToken
              );

            return next(authenticatedRequest);
          }

          return authService.refresh()
            .pipe(
              switchMap(session => {

                const retryRequest =
                  addAuthentication(
                    request.clone({
                      context: request.context.set(
                        AUTH_RETRY,
                        true
                      )
                    }),
                    session.accessToken
                  );

                return next(retryRequest);
              }),

              catchError(refreshError => {

                authService.clearSession();

                void router.navigate(
                  ['/login']
                );

                return throwError(
                  () => refreshError
                );
              })
            );
        })
      );
  };

/**
 * HTTP request'e gerekli authentication özelliklerini ekler.
 *
 * @param request Orijinal HTTP request.
 * @param accessToken Access token veya null.
 * @returns Authentication bilgileri eklenmiş request.
 */
function addAuthentication(
  request: HttpRequest<unknown>,
  accessToken: string | null
): HttpRequest<unknown> {

  if (!accessToken) {

    return request.clone({
      withCredentials: true
    });
  }

  return request.clone({

    withCredentials: true,

    setHeaders: {
      Authorization: `Bearer ${accessToken}`
    }
  });
}

/**
 * Otomatik refresh uygulanmaması gereken authentication
 * endpoint'lerini belirler.
 *
 * @param url HTTP request URL'sidir.
 * @returns Refresh atlanacaksa true.
 */
function shouldSkipAutomaticRefresh(
  url: string
): boolean {

  return [
    '/auth/login',
    '/auth/refresh',
    '/auth/forgot-password',
    '/auth/reset-password'
  ]
  .some(path => url.includes(path));
}

import { inject, Injectable } from '@angular/core';

import { AuthStoreService } from '../../auth/services/auth-store.service';

/**
 * Angular kullanıcı arayüzünde permission bazlı görünürlük
 * kontrollerini gerçekleştiren servistir.
 *
 * Bu servis bir güvenlik sınırı değildir.
 * Nihai yetki kontrolü backend tarafından yapılır.
 */
@Injectable({
  providedIn: 'root'
})
export class PermissionService {

  private readonly authStore =
    inject(AuthStoreService);

  /**
   * Kullanıcının belirtilen permission'a sahip olup
   * olmadığını kontrol eder.
   *
   * @param permission Permission kodudur.
   * @returns Kullanıcı permission'a sahipse true.
   */
  public has(
    permission: string
  ): boolean {

    const user =
      this.authStore.user();

    if (!user) {
      return false;
    }

    if (
      user.roles.includes('ADMIN')
    ) {
      return true;
    }

    return user.permissions.includes(
      permission
    );
  }

  /**
   * Kullanıcının verilen permission'lardan en az birine
   * sahip olup olmadığını kontrol eder.
   *
   * @param permissions Permission kodlarıdır.
   * @returns En az bir permission mevcutsa true.
   */
  public hasAny(
    ...permissions: string[]
  ): boolean {

    return permissions.some(
      permission => this.has(permission)
    );
  }

  /**
   * Kullanıcının verilen permission'ların tamamına
   * sahip olup olmadığını kontrol eder.
   *
   * @param permissions Permission kodlarıdır.
   * @returns Tüm permission'lar mevcutsa true.
   */
  public hasAll(
    ...permissions: string[]
  ): boolean {

    return permissions.every(
      permission => this.has(permission)
    );
  }
}

import { inject } from '@angular/core';

import {
  CanActivateFn,
  Router
} from '@angular/router';

import { PermissionService } from '../services/permission.service';

/**
 * Route seviyesinde permission kontrolü gerçekleştirir.
 *
 * Route data içerisinde:
 *
 * permissions: ['Users.View']
 *
 * şeklinde kullanılır.
 */
export const permissionGuard: CanActivateFn =
  route => {

    const permissionService =
      inject(PermissionService);

    const router =
      inject(Router);

    const permissions =
      route.data['permissions'] as string[] | undefined;

    const mode =
      route.data['permissionMode'] as
        'all' | 'any' | undefined;

    if (
      !permissions ||
      permissions.length === 0
    ) {
      return true;
    }

    const authorized =
      mode === 'any'
        ? permissionService.hasAny(
            ...permissions
          )
        : permissionService.hasAll(
            ...permissions
          );

    if (authorized) {
      return true;
    }

    return router.createUrlTree(
      ['/app']
    );
  };

} from 'primeng/password';

import {
  ButtonModule
} from 'primeng/button';

import {
  CardModule
} from 'primeng/card';

import {
  MessageModule
} from 'primeng/message';

import {
  AuthService
} from '../../../core/auth/services/auth.service';

/**
 * MSKBS kullanıcı giriş ekranını temsil eder.
 */
@Component({
  selector: 'app-login',
  standalone: true,

  imports: [
    CommonModule,
    ReactiveFormsModule,
    InputTextModule,
    PasswordModule,
    ButtonModule,
    CardModule,
    MessageModule
  ],

  templateUrl: './login.component.html',
  styleUrl: './login.component.scss'
})
export class LoginComponent {

  private readonly formBuilder =
    inject(FormBuilder);

  private readonly authService =
    inject(AuthService);

  private readonly router =
    inject(Router);

  private readonly activatedRoute =
    inject(ActivatedRoute);

  /**
   * Login HTTP isteğinin devam edip etmediğini belirtir.
   */
  public readonly loading =
    signal(false);

  /**
   * Kullanıcıya gösterilecek login hata mesajıdır.
   */
  public readonly errorMessage =
    signal<string | null>(null);

  /**
   * Login ekranındaki kullanıcı adı/e-posta ve parola alanlarını
   * içeren reactive form nesnesidir.
   */
  public readonly form =
    this.formBuilder.nonNullable.group({

      userNameOrEmail: [
        '',
        [
          Validators.required
        ]
      ],

      password: [
        '',
        [
          Validators.required
        ]
      ]
    });

  /**
   * Login formunu backend authentication endpoint'ine gönderir.
   */
  public submit(): void {

    this.errorMessage.set(null);

    if (this.form.invalid) {

      this.form.markAllAsTouched();

      return;
    }

    this.loading.set(true);

    const request =
      this.form.getRawValue();

    console.log(
      '[MSKBS AUTH] Login request gönderiliyor:',
      {
        userNameOrEmail:
          request.userNameOrEmail,

        /*
         * Parolayı console'a kesinlikle yazmıyoruz.
         */
        password: '********'
      }
    );

    this.authService.login(request)
      .pipe(
        finalize(() => {
          this.loading.set(false);
        })
      )
      .subscribe({

        next: response => {

          console.log(
            '[MSKBS AUTH] Login tamamlandı:',
            response
          );

          const returnUrl =
            this.activatedRoute
              .snapshot
              .queryParamMap
              .get('returnUrl');

          if (
            returnUrl &&
            this.isSafeReturnUrl(returnUrl)
          ) {

            void this.router.navigateByUrl(
              returnUrl
            );

            return;
          }

          void this.router.navigate(
            ['/app']
          );
        },

        error: (
          error: HttpErrorResponse
        ) => {

          console.error(
            '[MSKBS AUTH] Login error:',
            error
          );

          this.errorMessage.set(
            this.resolveErrorMessage(error)
          );
        }
      });
  }

  /**
   * Backend ProblemDetails response'undan kullanıcıya
   * gösterilecek uygun hata mesajını oluşturur.
   *
   * @param error HTTP hata nesnesidir.
   * @returns Kullanıcıya gösterilecek hata mesajı.
   */
  private resolveErrorMessage(
    error: HttpErrorResponse
  ): string {

    if (error.status === 423) {

      return (
        error.error?.detail ??
        'Hesabınız geçici olarak kilitlenmiştir.'
      );
    }

    if (error.status === 401) {

      return (
        error.error?.detail ??
        'Kullanıcı adı/e-posta veya parola hatalı.'
      );
    }

    if (error.status === 0) {

      return 'API sunucusuna ulaşılamıyor.';
    }

    return (
      error.error?.detail ??
      error.error?.title ??
      'Giriş sırasında beklenmeyen bir hata oluştu.'
    );
  }

  /**
   * returnUrl değerinin yalnızca Angular uygulaması içerisindeki
   * relative bir route olduğunu doğrular.
   *
   * @param url Kontrol edilecek route değeridir.
   * @returns Güvenli bir internal route ise true.
   */
  private isSafeReturnUrl(
    url: string
  ): boolean {

    return (
      url.startsWith('/') &&
      !url.startsWith('//')
    );
  }
}

/**
 * Backend ProblemDetails response'undan kullanıcıya gösterilecek
 * uygun hata mesajını oluşturur.
 *
 * @param error HTTP hata nesnesidir.
 * @returns Kullanıcıya gösterilecek hata mesajı.
 */
private resolveErrorMessage(
  error: HttpErrorResponse
): string {

  const problem =
    error.error as ProblemDetails | null;

  if (error.status === 423) {
    return (
      problem?.detail ??
      'Hesabınız geçici olarak kilitlenmiştir.'
    );
  }

  if (error.status === 401) {
    return (
      problem?.detail ??
      'Kullanıcı adı/e-posta veya parola hatalı.'
    );
  }

  if (error.status === 400) {
    return (
      problem?.detail ??
      'Gönderilen bilgiler geçerli değildir.'
    );
  }

  if (error.status === 0) {
    return 'API sunucusuna ulaşılamıyor.';
  }

  return (
    problem?.detail ??
    problem?.title ??
    'Giriş sırasında beklenmeyen bir hata oluştu.'
  );
}

3. Backend refresh cookie ayarını kontrol edelim
Angular refresh token'ı görmeyecek.
Backend login sırasında cookie oluşturmalı.
AuthController içerisinde muhtemelen buna benzer bir yapı yazmıştık.
Cookie oluşturmayı merkezi metoda alalım.
Örneğin:
/// <summary>
/// Refresh token değerini güvenli HttpOnly cookie olarak
/// HTTP response'a ekler.
/// </summary>
/// <param name="refreshToken">
/// Kullanıcıya gönderilecek açık refresh token değeridir.
/// </param>
/// <param name="expiresAt">
/// Refresh token'ın UTC sona erme zamanıdır.
/// </param>
private void SetRefreshTokenCookie(
    string refreshToken,
    DateTime expiresAt)
{
    Response.Cookies.Append(
        "MSKBS.RefreshToken",
        refreshToken,
        new CookieOptions
        {
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            Expires = expiresAt,
            Path = "/api/auth"
        });
}

Burada:
HttpOnly = true

JavaScript token'ı okuyamaz.
Secure = true

HTTPS zorlar.
SameSite = Strict

CSRF yüzeyini azaltır.
Path = /api/auth

refresh cookie gereksiz yere bütün API endpointlerine gitmez.
Login endpoint'inde:
var result = await _mediator.Send(command);

SetRefreshTokenCookie(
    result.RefreshToken,
    result.RefreshTokenExpiresAt);

return Ok(result.Response);

mantığı olmalı.
Refresh token response body'ye dönmemeli.
4. Refresh endpoint cookie'yi buradan okuyacak
Örneğin:
/// <summary>
/// HttpOnly refresh cookie kullanılarak yeni access token
/// ve refresh token üretir.
/// </summary>
/// <param name="cancellationToken">
/// Request iptal token'ıdır.
/// </param>
/// <returns>
/// Yenilenmiş authentication session bilgisini döndürür.
/// </returns>
[HttpPost("refresh")]
[AllowAnonymous]
public async Task<ActionResult<LoginResponse>> Refresh(
    CancellationToken cancellationToken)
{
    if (!Request.Cookies.TryGetValue(
        "MSKBS.RefreshToken",
        out var refreshToken))
    {
        throw new InvalidRefreshTokenException();
    }

    var command = new RefreshTokenCommand
    {
        RefreshToken = refreshToken
    };

    var result = await _mediator.Send(
        command,
        cancellationToken);

    SetRefreshTokenCookie(
        result.RefreshToken,
        result.RefreshTokenExpiresAt);

    return Ok(result.Response);
}

Sende command/property isimleri biraz farklıysa mevcut olanı bozma; önemli olan mantık:
Cookie
 ↓
Refresh handler
 ↓
rotation
 ↓
yeni access token
+
yeni refresh cookie

/**
 * Kullanıcının e-posta adresine parola sıfırlama
 * bağlantısı gönderilmesini talep eder.
 *
 * Backend kullanıcı mevcut olsa da olmasa da aynı cevabı
 * döndürerek kullanıcı hesabı keşfini engeller.
 *
 * @param request Kullanıcının e-posta adresini içeren istektir.
 * @returns Forgot password HTTP işlemi.
 */
public forgotPassword(
  request: ForgotPasswordRequest
): Observable<void> {

  return this.http.post<void>(
    `${this.apiBaseUrl}/auth/forgot-password`,
    request
  );
}

/**
 * E-posta bağlantısından alınan reset token kullanılarak
 * kullanıcının parolasını yeniler.
 *
 * @param request Token ve yeni parola bilgilerini içeren istektir.
 * @returns Reset password HTTP işlemi.
 */
public resetPassword(
  request: ResetPasswordRequest
): Observable<void> {

  return this.http.post<void>(
    `${this.apiBaseUrl}/auth/reset-password`,
    request
  )
  .pipe(
    tap(() => {
      /*
       * Backend parola sıfırlama sonrasında SecurityStamp
       * değiştirdiği ve refresh token'ları revoke ettiği için
       * istemci authentication state'i de temizlenir.
       */
      this.store.clearSession();
    })
  );
}

/**
 * Oturum açmış kullanıcının mevcut parolasını değiştirir.
 *
 * Başarılı işlemden sonra backend SecurityStamp değerini
 * değiştirir ve refresh token'ları revoke eder. Bu nedenle
 * kullanıcı yeniden login olmalıdır.
 *
 * @param request Mevcut ve yeni parola bilgilerini içerir.
 * @returns Change password HTTP işlemi.
 */
public changePassword(
  request: ChangePasswordRequest
): Observable<void> {

  return this.http.post<void>(
    `${this.apiBaseUrl}/auth/change-password`,
    request,
    {
      withCredentials: true
    }
  )
  .pipe(
    tap(() => {
      this.store.clearSession();
    })
  );
}

  from '../../../core/auth/services/auth.service';

/**
 * Kullanıcının e-posta ile aldığı parola sıfırlama token'ı
 * üzerinden yeni parola belirlediği ekranı temsil eder.
 */
@Component({
  selector: 'app-reset-password',
  standalone: true,

  imports: [
    ReactiveFormsModule,
    RouterLink,
    PasswordModule,
    ButtonModule,
    CardModule,
    MessageModule
  ],

  templateUrl: './reset-password.component.html',
  styleUrl: './reset-password.component.scss'
})
export class ResetPasswordComponent {

  private readonly formBuilder =
    inject(FormBuilder);

  private readonly authService =
    inject(AuthService);

  private readonly activatedRoute =
    inject(ActivatedRoute);

  private readonly router =
    inject(Router);

  /**
   * URL üzerinden alınan parola sıfırlama token'ıdır.
   */
  public readonly token =
    this.activatedRoute.snapshot.queryParamMap.get('token');

  /**
   * HTTP isteğinin devam edip etmediğini belirtir.
   */
  public readonly loading =
    signal(false);

  /**
   * Kullanıcıya gösterilecek hata mesajıdır.
   */
  public readonly errorMessage =
    signal<string | null>(null);

  /**
   * Parola sıfırlama işleminin başarıyla tamamlanıp
   * tamamlanmadığını belirtir.
   */
  public readonly completed =
    signal(false);

  /**
   * Yeni parola ve parola tekrar alanlarını içeren
   * reset password formudur.
   */
  public readonly form =
    this.formBuilder.nonNullable.group(
      {
        newPassword: [
          '',
          [
            Validators.required
          ]
        ],

        confirmPassword: [
          '',
          [
            Validators.required
          ]
        ]
      },
      {
        validators: passwordsMatchValidator()
      }
    );

  /**
   * Yeni parola bilgisini backend'e göndererek
   * kullanıcının parolasını sıfırlar.
   */
  public submit(): void {

    this.errorMessage.set(null);

    if (!this.token) {

      this.errorMessage.set(
        'Parola sıfırlama bağlantısı geçersizdir.'
      );

      return;
    }

    if (this.form.invalid) {

      this.form.markAllAsTouched();

      return;
    }

    this.loading.set(true);

    const value =
      this.form.getRawValue();

    this.authService
      .resetPassword({
        token: this.token,
        newPassword: value.newPassword,
        confirmPassword: value.confirmPassword
      })
      .pipe(
        finalize(() => {
          this.loading.set(false);
        })
      )
      .subscribe({

        next: () => {

          this.completed.set(true);

          /*
           * Kullanıcı mesajı okuyabilsin diye kısa bir
           * gecikmeyle login ekranına yönlendiriyoruz.
           */
          setTimeout(() => {

            void this.router.navigate(
              ['/login']
            );

          }, 2000);
        },

        error: (
          error: HttpErrorResponse
        ) => {

          console.error(
            '[MSKBS AUTH] Reset password error:',
            error
          );

          this.errorMessage.set(
            error.error?.detail ??
            'Parola sıfırlama bağlantısı geçersiz veya süresi dolmuş olabilir.'
          );
        }
      });
  }
}

/**
 * Yeni parola ve parola tekrar alanlarının
 * birbiriyle aynı olmasını doğrular.
 *
 * @returns Angular form validator fonksiyonu.
 */
function passwordsMatchValidator(): ValidatorFn {

  return (
    control: AbstractControl
  ): ValidationErrors | null => {

    const newPassword =
      control.get('newPassword')?.value;

    const confirmPassword =
      control.get('confirmPassword')?.value;

    if (
      !newPassword ||
      !confirmPassword
    ) {
      return null;
    }

    return newPassword === confirmPassword
      ? null
      : {
          passwordMismatch: true
        };
  };
}

import {
  Component,
  inject,
  signal
} from '@angular/core';

import {
  CommonModule
} from '@angular/common';

import {
  FormBuilder,
  ReactiveFormsModule,
  Validators
} from '@angular/forms';

import {
  HttpErrorResponse
} from '@angular/common/http';

import {
  ActivatedRoute,
  Router,
  RouterLink
} from '@angular/router';

import {
  finalize
} from 'rxjs';

import {
  InputTextModule
} from 'primeng/inputtext';

import {
  PasswordModule
} from 'primeng/password';

import {
  ButtonModule
} from 'primeng/button';

import {
  CardModule
} from 'primeng/card';

import {
  MessageModule
} from 'primeng/message';

import {
  AuthService
} from '../../../core/auth/services/auth.service';

import {
  ProblemDetails
} from '../../../core/models/problem-details.model';

/**
 * MSKBS kullanıcı giriş ekranını temsil eder.
 *
 * Kullanıcı adı veya e-posta adresi ile giriş yapılmasını sağlar.
 */
@Component({
  selector: 'app-login',
  standalone: true,

  imports: [
    CommonModule,
    ReactiveFormsModule,
    RouterLink,
    InputTextModule,
    PasswordModule,
    ButtonModule,
    CardModule,
    MessageModule
  ],

  templateUrl: './login.component.html',
  styleUrl: './login.component.scss'
})
export class LoginComponent {

  private readonly formBuilder =
    inject(FormBuilder);

  private readonly authService =
    inject(AuthService);

  private readonly router =
    inject(Router);

  private readonly activatedRoute =
    inject(ActivatedRoute);

  /**
   * Login isteğinin devam edip etmediğini belirtir.
   */
  public readonly loading =
    signal(false);

  /**
   * Kullanıcıya gösterilecek login hata mesajıdır.
   */
  public readonly errorMessage =
    signal<string | null>(null);

  /**
   * Login ekranında gösterilecek bilgilendirme mesajıdır.
   */
  public readonly informationMessage =
    signal<string | null>(null);

  /**
   * Login form modelidir.
   */
  public readonly form =
    this.formBuilder.nonNullable.group({

      userNameOrEmail: [
        '',
        [
          Validators.required
        ]
      ],

      password: [
        '',
        [
          Validators.required
        ]
      ]
    });

  /**
   * Login ekranı başlangıç kontrollerini gerçekleştirir.
   */
  public constructor() {

    const passwordChanged =
      this.activatedRoute
        .snapshot
        .queryParamMap
        .get('passwordChanged');

    const passwordReset =
      this.activatedRoute
        .snapshot
        .queryParamMap
        .get('passwordReset');

    const loggedOut =
      this.activatedRoute
        .snapshot
        .queryParamMap
        .get('loggedOut');

    if (passwordChanged === 'true') {

      this.informationMessage.set(
        'Parolanız başarıyla değiştirildi. Yeni parolanızla giriş yapabilirsiniz.'
      );

      return;
    }

    if (passwordReset === 'true') {

      this.informationMessage.set(
        'Parolanız başarıyla sıfırlandı. Yeni parolanızla giriş yapabilirsiniz.'
      );

      return;
    }

    if (loggedOut === 'true') {

      this.informationMessage.set(
        'Oturumunuz başarıyla kapatıldı.'
      );
    }
  }

  /**
   * Login formunu backend authentication endpoint'ine gönderir.
   */
  public submit(): void {

    this.errorMessage.set(null);

    if (this.form.invalid) {

      this.form.markAllAsTouched();

      return;
    }

    this.loading.set(true);

    const request =
      this.form.getRawValue();

    console.log(
      '[MSKBS AUTH] Login request:',
      {
        userNameOrEmail:
          request.userNameOrEmail,

        password:
          '********'
      }
    );

    this.authService
      .login(request)
      .pipe(
        finalize(() => {
          this.loading.set(false);
        })
      )
      .subscribe({

        next: response => {

          console.log(
            '[MSKBS AUTH] Login başarılı:',
            response
          );

          const returnUrl =
            this.activatedRoute
              .snapshot
              .queryParamMap
              .get('returnUrl');

          if (
            returnUrl &&
            this.isSafeReturnUrl(returnUrl)
          ) {

            void this.router.navigateByUrl(
              returnUrl
            );

            return;
          }

          void this.router.navigate(
            ['/app']
          );
        },

        error: (
          error: HttpErrorResponse
        ) => {

          console.error(
            '[MSKBS AUTH] Login hatası:',
            error
          );

          this.errorMessage.set(
            this.resolveErrorMessage(error)
          );
        }
      });
  }

  /**
   * Backend ProblemDetails cevabından kullanıcıya
   * gösterilecek uygun hata mesajını üretir.
   *
   * @param error HTTP hata nesnesidir.
   * @returns Kullanıcıya gösterilecek hata mesajı.
   */
  private resolveErrorMessage(
    error: HttpErrorResponse
  ): string {

    const problem =
      error.error as ProblemDetails | null;

    if (error.status === 423) {

      return (
        problem?.detail ??
        'Hesabınız geçici olarak kilitlenmiştir.'
      );
    }

    if (error.status === 401) {

      return (
        problem?.detail ??
        'Kullanıcı adı/e-posta veya parola hatalı.'
      );
    }

    if (error.status === 400) {

      return (
        problem?.detail ??
        'Gönderilen bilgiler geçerli değildir.'
      );
    }

    if (error.status === 0) {

      return 'API sunucusuna ulaşılamıyor.';
    }

    return (
      problem?.detail ??
      problem?.title ??
      'Giriş sırasında beklenmeyen bir hata oluştu.'
    );
  }

  /**
   * returnUrl değerinin yalnızca Angular uygulaması
   * içerisindeki route'lara yönlenmesini sağlar.
   *
   * @param url Kontrol edilecek URL değeridir.
   * @returns Güvenli ise true.
   */
  private isSafeReturnUrl(
    url: string
  ): boolean {

    return (
      url.startsWith('/') &&
      !url.startsWith('//')
    );
  }
}

Final hali:

/**
 * Mevcut cihazdaki refresh token'ı revoke eder ve
 * local authentication state'ini temizler.
 *
 * @returns Logout işlemini temsil eden Observable.
 */
public logout(): Observable<void> {

  return this.http.post<void>(
    `${this.apiBaseUrl}/auth/logout`,
    {},
    {
      withCredentials: true
    }
  )
  .pipe(
    finalize(() => {

      this.store.clearSession();
    })
  );
}

/**
 * Kullanıcının bütün refresh token'larını revoke eder
 * ve local authentication state'ini temizler.
 *
 * Backend ayrıca SecurityStamp değerini değiştirerek
 * mevcut access token'ların da geçersiz hale gelmesini sağlar.
 *
 * @returns Logout-all işlemini temsil eden Observable.
 */
public logoutAll(): Observable<void> {

  return this.http.post<void>(
    `${this.apiBaseUrl}/auth/logout-all`,
    {},
    {
      withCredentials: true
    }
  )
  .pipe(
    finalize(() => {

      this.store.clearSession();
    })
  );
}

import {
  computed,
  inject,
  Injectable
} from '@angular/core';

import {
  PermissionCodes
} from '../../core/permissions/constants/permission-codes';

import {
  PermissionService
} from '../../core/permissions/services/permission.service';

import {
  NavigationItem
} from '../models/navigation-item.model';

/**
 * MSKBS ana uygulama menüsünü merkezi olarak tanımlar
 * ve mevcut kullanıcının permission bilgilerine göre filtreler.
 */
@Injectable({
  providedIn: 'root'
})
export class NavigationService {

  private readonly permissionService =
    inject(PermissionService);

  /**
   * MSKBS uygulamasındaki ana navigasyon tanımlarını içerir.
   *
   * Feature ekranları geliştirildikçe route değerleri
   * bu merkezi liste üzerinden tanımlanacaktır.
   */
  private readonly navigationItems:
    readonly NavigationItem[] = [

    {
      label: 'Ana Sayfa',
      icon: 'pi pi-home',
      route: '/app'
    },

    {
      label: 'Kullanıcılar',
      icon: 'pi pi-users',

      requiredPermissions: [
        PermissionCodes.Users.View
      ]

      /*
       * Kullanıcı ekranını geliştirdiğimizde:
       *
       * route: '/app/users'
       */
    },

    {
      label: 'Roller',
      icon: 'pi pi-shield',

      requiredPermissions: [
        PermissionCodes.Roles.View
      ]
    },

    {
      label: 'Organizasyon',
      icon: 'pi pi-sitemap',

      requiredPermissions: [
        PermissionCodes.Organizations.View
      ]
    },

    {
      label: 'İncelemeler',
      icon: 'pi pi-search',

      requiredPermissions: [
        PermissionCodes.Investigations.View
      ]
    },

    {
      label: 'Raporlar',
      icon: 'pi pi-chart-bar',

      requiredPermissions: [
        PermissionCodes.Reports.View
      ]
    }
  ];

  /**
   * Mevcut kullanıcının permission bilgilerine göre
   * görüntüleyebileceği navigasyon elemanlarını döndürür.
   *
   * Kullanıcının authentication state'i değiştiğinde
   * Signal otomatik olarak yeniden hesaplanır.
   */
  public readonly visibleItems =
    computed<readonly NavigationItem[]>(
      () =>
        this.filterItems(
          this.navigationItems
        )
    );

  /**
   * Navigasyon listesini permission kurallarına göre filtreler.
   *
   * @param items Filtrelenecek navigasyon elemanlarıdır.
   * @returns Kullanıcının görüntüleyebileceği menü elemanları.
   */
  private filterItems(
    items: readonly NavigationItem[]
  ): NavigationItem[] {

    const result: NavigationItem[] = [];

    for (const item of items) {

      if (!this.canView(item)) {
        continue;
      }

      const children =
        item.children
          ? this.filterItems(item.children)
          : undefined;

      /*
       * Sadece alt menülerden oluşan bir başlıkta
       * görüntülenecek hiçbir child kalmadıysa parent da gizlenir.
       */
      if (
        item.children &&
        !item.route &&
        (!children || children.length === 0)
      ) {
        continue;
      }

      result.push({
        ...item,
        children
      });
    }

    return result;
  }

  /**
   * Belirtilen navigasyon elemanının mevcut kullanıcı tarafından
   * görüntülenip görüntülenemeyeceğini kontrol eder.
   *
   * @param item Kontrol edilecek navigasyon elemanıdır.
   * @returns Kullanıcı menüyü görebiliyorsa true.
   */
  private canView(
    item: NavigationItem
  ): boolean {

    const permissions =
      item.requiredPermissions;

    if (
      !permissions ||
      permissions.length === 0
    ) {
      return true;
    }

    if (
      item.permissionMode === 'any'
    ) {

      return this.permissionService.hasAny(
        ...permissions
      );
    }

    return this.permissionService.hasAll(
      ...permissions
    );
  }
}

        firstName
          ? firstName.charAt(0)
          : '';

      const lastInitial =
        lastName
          ? lastName.charAt(0)
          : '';

      return (
        firstInitial + lastInitial
      ).toUpperCase();
    });

  /**
   * Kullanıcı işlemleri popup menüsünü temsil eder.
   */
  public readonly userMenuItems:
    MenuItem[] = [

    {
      label: 'Parola Değiştir',
      icon: 'pi pi-key',

      command: () => {

        void this.router.navigate(
          ['/app/change-password']
        );
      }
    },

    {
      separator: true
    },

    {
      label: 'Çıkış Yap',
      icon: 'pi pi-sign-out',

      command: () => {

        this.logout();
      }
    }
  ];

  /**
   * Sidebar açma/kapama isteği oluşturur.
   */
  public toggleSidebar(): void {

    this.sidebarToggle.emit();
  }

  /**
   * Kullanıcının mevcut cihazdaki oturumunu kapatır
   * ve login ekranına yönlendirir.
   */
  private logout(): void {

    this.authService.logout()
      .subscribe({

        next: () => {

          void this.router.navigate(
            ['/login'],
            {
              queryParams: {
                loggedOut: true
              }
            }
          );
        },

        error: error => {

          console.error(
            '[MSKBS AUTH] Logout error:',
            error
          );

          void this.router.navigate(
            ['/login']
          );
        }
      });
  }
}

namespace MSKBS.Application.Common.Models;

/// <summary>
/// Sayfalı sorguların standart response modelini temsil eder.
/// </summary>
/// <typeparam name="T">
/// Sayfalama içerisinde döndürülecek kayıt tipidir.
/// </typeparam>
public sealed class PagedResult<T>
{
    /// <summary>
    /// İstenen sayfaya ait kayıtları içerir.
    /// </summary>
    public IReadOnlyCollection<T> Items { get; init; } = [];

    /// <summary>
    /// Uygulanan filtrelere uyan toplam kayıt sayısını belirtir.
    /// </summary>
    public int TotalCount { get; init; }

    /// <summary>
    /// Mevcut sayfa numarasını belirtir.
    /// </summary>
    public int PageNumber { get; init; }

    /// <summary>
    /// Bir sayfada döndürülecek maksimum kayıt sayısını belirtir.
    /// </summary>
    public int PageSize { get; init; }

    /// <summary>
    /// Toplam sayfa sayısını hesaplar.
    /// </summary>
    public int TotalPages =>
        PageSize <= 0
            ? 0
            : (int)Math.Ceiling(
                TotalCount / (double)PageSize);
}

5. GetUsersQueryHandler
Dosya:

MSKBS.Application/Users/Queries/GetUsers/GetUsersQueryHandler.cs
using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Application.Common.Models;

namespace MSKBS.Application.Users.Queries.GetUsers;

/// <summary>
/// Kullanıcı listeleme ve sorgulama işlemini gerçekleştirir.
/// </summary>
internal sealed class GetUsersQueryHandler
    : IRequestHandler<
        GetUsersQuery,
        PagedResult<UserListItemResponse>>
{
    private readonly IMSKBSDbContext _dbContext;

    /// <summary>
    /// Kullanıcı sorgu handler'ını oluşturur.
    /// </summary>
    /// <param name="dbContext">
    /// MSKBS uygulama veritabanı context abstraction'ıdır.
    /// </param>
    public GetUsersQueryHandler(
        IMSKBSDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    /// <summary>
    /// Kullanıcıları filtreleyerek ve sayfalayarak döndürür.
    /// </summary>
    /// <param name="request">
    /// Kullanıcı sorgu kriterlerini içerir.
    /// </param>
    /// <param name="cancellationToken">
    /// İşlemin iptal token'ıdır.
    /// </param>
    /// <returns>
    /// Sayfalanmış kullanıcı listesini döndürür.
    /// </returns>
    public async Task<PagedResult<UserListItemResponse>> Handle(
        GetUsersQuery request,
        CancellationToken cancellationToken)
    {
        IQueryable<Domain.Entities.Identity.User> query =
            _dbContext.Users
                .AsNoTracking();

        /*
         * Normal sorgularda ActiveFilter ve SoftDeleteFilter
         * otomatik çalışır.
         *
         * Admin pasif kullanıcıları da görmek istediğinde yalnızca
         * ActiveFilter kapatılır. SoftDeleteFilter çalışmaya devam eder.
         *
         * Buradaki string EF Core modelinde tanımladığımız named
         * filter adı ile birebir aynıdır.
         */
        if (request.IncludeInactive)
        {
            query = query.IgnoreQueryFilters(
                ["ActiveFilter"]);
        }

        if (!string.IsNullOrWhiteSpace(request.Search))
        {
            var search =
                request.Search
                    .Trim()
                    .ToLowerInvariant();

            query = query.Where(user =>
                user.UserName.ToLower().Contains(search) ||
                user.Email.ToLower().Contains(search) ||
                user.FirstName.ToLower().Contains(search) ||
                user.LastName.ToLower().Contains(search));
        }

        var totalCount =
            await query.CountAsync(
                cancellationToken);

        var skip =
            (request.PageNumber - 1) *
            request.PageSize;

        var items =
            await query
                .OrderBy(user => user.FirstName)
                .ThenBy(user => user.LastName)
                .ThenBy(user => user.UserName)
                .Skip(skip)
                .Take(request.PageSize)
                .Select(user =>
                    new UserListItemResponse
                    {
                        Id = user.Id,

                        UserName =
                            user.UserName,

                        FirstName =
                            user.FirstName,

                        LastName =
                            user.LastName,

                        FullName =
                            user.FirstName + " " +
                            user.LastName,

                        Email =
                            user.Email,

                        PhoneNumber =
                            user.PhoneNumber,

                        OrganizationUnitId =
                            user.OrganizationUnitId,

                        OrganizationUnitCode =
                            user.OrganizationUnit != null
                                ? user.OrganizationUnit.Code
                                : null,

                        OrganizationUnitName =
                            user.OrganizationUnit != null
                                ? user.OrganizationUnit.Name
                                : null,

                        Roles =
                            user.UserRoles
                                .Where(userRole =>
                                    !userRole.IsDeleted &&
                                    userRole.IsActive &&
                                    !userRole.Role.IsDeleted &&
                                    userRole.Role.IsActive)
                                .OrderBy(userRole =>
                                    userRole.Role.Name)
                                .Select(userRole =>
                                    userRole.Role.Code)
                                .ToArray(),

                        IsActive =
                            user.IsActive,

                        IsLocked =
                            user.IsLocked,

                        CreatedDate =
                            user.CreatedDate,

                        Version =
                            user.Version
                    })
                .ToListAsync(
                    cancellationToken);

        return new PagedResult<UserListItemResponse>
        {
            Items = items,
            TotalCount = totalCount,
            PageNumber = request.PageNumber,
            PageSize = request.PageSize
        };
    }
}

/**

@Injectable({
  providedIn: 'root'
})
export class UserService {

  private readonly http =
    inject(HttpClient);

  private readonly apiBaseUrl =
    inject(API_BASE_URL);

  /**
   * Kullanıcıları filtrelenmiş ve sayfalanmış
   * şekilde backend'den getirir.
   *
   * @param query Kullanıcı sorgu kriterleridir.
   * @returns Sayfalanmış kullanıcı listesi.
   */
  public getUsers(
    query: UserQuery
  ): Observable<PagedResult<UserListItem>> {

    let params =
      new HttpParams()
        .set(
          'pageNumber',
          query.pageNumber
        )
        .set(
          'pageSize',
          query.pageSize
        )
        .set(
          'includeInactive',
          query.includeInactive
        );

    const search =
      query.search?.trim();

    if (search) {

      params = params.set(
        'search',
        search
      );
    }

    return this.http.get<
      PagedResult<UserListItem>
    >(
      `${this.apiBaseUrl}/users`,
      {
        params
      }
    );
  }

 */
@Component({
  selector: 'app-user-list',
  standalone: true,

  imports: [
    DatePipe,
    ReactiveFormsModule,
    TableModule,
    ButtonModule,
    InputTextModule,
    TagModule,
    CheckboxModule,
    PaginatorModule,
    MessageModule
  ],

  templateUrl: './user-list.component.html',
  styleUrl: './user-list.component.scss'
})
export class UserListComponent
  implements OnInit {

  private readonly userService =
    inject(UserService);

  private readonly formBuilder =
    inject(FormBuilder);

  /**
   * Kullanıcı listesini içerir.
   */
  public readonly users =
    signal<UserListItem[]>([]);

  /**
   * Filtrelere uyan toplam kullanıcı sayısıdır.
   */
  public readonly totalCount =
    signal(0);

  /**
   * Aktif sayfa numarasıdır.
   */
  public readonly pageNumber =
    signal(1);

  /**
   * Sayfa başına gösterilecek kayıt sayısıdır.
   */
  public readonly pageSize =
    signal(20);

  /**
   * Kullanıcı sorgusunun devam edip etmediğini belirtir.
   */
  public readonly loading =
    signal(false);

  /**
   * Listeleme sırasında kullanıcıya gösterilecek
   * hata mesajıdır.
   */
  public readonly errorMessage =
    signal<string | null>(null);

  /**
   * Kullanıcı arama ve filtreleme formudur.
   */
  public readonly filterForm =
    this.formBuilder.nonNullable.group({

      search: [''],

      includeInactive: [false]
    });

  /**
   * Component başlangıcında ilk kullanıcı listesini yükler.
   */
  public ngOnInit(): void {

    this.loadUsers();
  }

  /**
   * Kullanıcı listesini mevcut filtre ve sayfalama
   * kriterlerine göre backend'den getirir.
   */
  public loadUsers(): void {

    this.loading.set(true);
    this.errorMessage.set(null);

    const filters =
      this.filterForm.getRawValue();

    this.userService
      .getUsers({
        search:
          filters.search,

        includeInactive:
          filters.includeInactive,

        pageNumber:
          this.pageNumber(),

        pageSize:
          this.pageSize()
      })
      .pipe(
        finalize(() => {

          this.loading.set(false);
        })
      )
      .subscribe({

        next: response => {

          this.users.set(
            response.items
          );

          this.totalCount.set(
            response.totalCount
          );
        },

        error: (
          error: HttpErrorResponse
        ) => {

          console.error(
            '[MSKBS USERS] User query error:',
            error
          );

          this.users.set([]);

          this.totalCount.set(0);

          this.errorMessage.set(
            error.error?.detail ??
            'Kullanıcılar yüklenirken bir hata oluştu.'
          );
        }
      });
  }

  /**
   * Arama butonuna basıldığında sayfayı başa alır
   * ve kullanıcı sorgusunu tekrar çalıştırır.
   */
  public search(): void {

    this.pageNumber.set(1);

    this.loadUsers();
  }

  /**
   * Kullanıcı filtrelerini varsayılan değerlere döndürür
   * ve listeyi yeniden yükler.
   */
  public clearFilters(): void {

    this.filterForm.reset({
      search: '',
      includeInactive: false
    });

    this.pageNumber.set(1);

    this.loadUsers();
  }

  /**
   * PrimeNG paginator değişikliğini işler.
   *
   * @param event PrimeNG paginator state bilgisidir.
   */
  public onPageChange(
    event: PaginatorState
  ): void {

    const rows =
      event.rows ?? this.pageSize();

    const first =
      event.first ?? 0;

    const newPageNumber =
      Math.floor(first / rows) + 1;

    this.pageSize.set(rows);

    this.pageNumber.set(
      newPageNumber
    );

    this.loadUsers();
  }
}
