  openCam() {
    this.camOpen = true;

  cameraoff() {
    const videoElem = <any>document.getElementsByTagName("video")[0];
    if (videoElem) {
      const stream = videoElem.srcObject;
      if (stream) {

        stream.getTracks().forEach((track) => {
          track.stop();
        });

        videoElem.srcObject = null;
      }
    }
    if (this.viewModel.TempBinaryDataObjectId) {
      this.viewModel.TempBinaryDataObjectId = null;
    }
  }

      thumbnailHeight: 100,
      thumbnailWidth: 100,
      uploadUrl: this.fileUploadService.getUploadImageUrl(),

        this.senderOtherReq = true;
      }
    } else {

    }

      this.controlOfRequiredMessage += "Kuvvet bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.LastNominationDateRequired == false && this.viewModel.nasbi == null) {
      this.controlOfRequiredMessage += "Nasbi bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.MilitaryClassRequired == false && this.viewModel.sinif == null) {
      this.controlOfRequiredMessage += "Sınıf bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.MilitaryOfficeRequired == false && this.viewModel.kayitliOlduguAskerlikSubesi == null) {
      this.controlOfRequiredMessage += "Askerlik Şubesi bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.MilitarySchoolRequired == false && this.viewModel.okulu == null) {
      this.controlOfRequiredMessage += "Okulu bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.MilitaryUnitRequired == false && this.viewModel.birlik == null) {
      this.controlOfRequiredMessage += "Birlik bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.RankRequired == false && this.viewModel.rutbe == null) {
      this.controlOfRequiredMessage += "Rütbe bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.RegistryNumberRequired == false && this.viewModel.sicilNo == null) {
      this.controlOfRequiredMessage += "Sicil Numarası bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.viewModel.muayeneyeGonderenMakam == null) //this.SenderChaireRequired == false && 
    {
      this.controlOfRequiredMessage += "Muayeneye Gönderen Makam bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }

    if (this.NthAdmissionRequired == false && this.viewModel.kacinciIslem == null) {
      this.controlOfRequiredMessage += "Kaçıncı işlem bilgisi boş geçilemez.";
      this.controlOfRequired = true;
    }
    if (this.DispatchDocumentDateRequired == false && this.viewModel.sevkEvrakTarihi == null) {
      this.controlOfRequiredMessage += "Sevk evrak tarihi bilgisi boş geçilemez.";
      this.controlOfRequired = true;

      this.controlOfRequiredMessage += "Sevk evrak sayısı bilgisi boş geçilemez.";

  save() {

    if (!this.form.valid) {
      if (this.form) {
        this.messageService.showError('Zorunlu (kırmızı) alanlar boş geçilemez');
        return;
      }
    }

    this.controlOfRequiredFiles();
    if (this.controlOfRequired) {
      this.messageService.showError(this.controlOfRequiredMessage);
      return;
    }


    if (this.senderOtherReq) {
      if (this.viewModel.muayeneyeGonderenMakamAciklama == null || this.viewModel.muayeneyeGonderenMakamAciklama == "") {
        this.messageService.showError('Muayeneye gönderen makam diğer için açıklama boş geçilemez.');
        return;
      }
    }

    if (this.senderOtherMilitaryReq) {
      if (this.viewModel.birlikAciklama == null || this.viewModel.birlikAciklama == "") {
        this.messageService.showError('Birlik açıklama diğer için açıklama boş geçilemez.');
        return;
      }
    }

    const that = this;

     //this.viewModel.telefonNo = 5415821575;
    this.dispatchService.newDispatchPhoto(this.viewModel).toPromise().then(result => {

      if (result.sonucKodu === MSKBSResultCodes.SuccessResultCode) {

    const message =
        `${result.sonucKodu}-${result.sonucMesaji}. ` +
        `${result.sevkSiraNo} sevk numarası ile işlem kaydedilmiştir.`;

    that.messageService.showInfo(message);

    // DispatchFile GUID'lerini grid'e bas
    //this.gridDataSourceDispatchFile = (result.dispatchFileIds || []).map(id => ({
    //dispatchFileId: id}));

    this.viewModel = new SevkYeniGirisFoto();
    //this.viewModel.dispatchFiles = [];
    this.gridDataSourceDispatchFile = [];
    this.dispatchFileValue = [];
   
}else {
        const message = `${result.sonucKodu}-${result.sonucMesaji}.`;
        that.messageService.showError(message);
      }
    },
      error => {
        if (error && error.message) {
          that.messageService.showError(error.message);
          if (error.status == 401 || error.status == 0) {
            const uri = `/kimlik/giris`;
            this.router.navigate([uri]);
          }
        } else {
          if (error) {
            that.messageService.showError(error.toString());
          }
        }
      });
  }

  public queryPerson() {
    this.sevkBilgileriniBosalt();
    const that = this;
    if (this.isTcOrForeign === this.identificationnumbers[0]) {
      if (that.uniqueRefno === "" || typeof that.uniqueRefno === 'undefined' || that.uniqueRefno.length != 11) {
        this.messageService.showWarning(`Geçerli T.C. kimlik numarası girmelisiniz!`);
        //confirm(`Geçerli T.C. kimlik numarası girmelisiniz!`);
        that.uniqueRefno = undefined;
      }
      if (that.uniqueRefno !== "" && typeof that.uniqueRefno !== 'undefined' && that.uniqueRefno.length === 11) {
        this.personGetInput.uniqueRefno = Number(that.uniqueRefno);
        this.personGetInput.birthDate = that.dogumTarihi;//this.viewModel.dogumTarihi;
        this.personService.getPerson(this.personGetInput).toPromise().then((personRes: PersonResult) => {
          if (personRes && personRes.person) {
            const person = personRes.person;
            that.viewModel.tcKimlikNo = person.uniqueRefno;
            that.viewModel.adi = person.name;
            that.viewModel.soyadi = person.surname;
            that.viewModel.anneAdi = person.motherName;
            that.viewModel.babaAdi = person.fatherName;
            that.viewModel.cinsiyet = person.gender;
            that.viewModel.dogumTarihi = person.birthDate;
            that.viewModel.adres = person.address;

            that.viewModel.kuvvet = person.forcesCommand;
            that.viewModel.sinif = person.militaryClass;
            that.viewModel.rutbe = person.rank;
            that.viewModel.birlik = person.militaryUnit;
            that.viewModel.birlikAciklama = person.militaryUnitDescription;
            that.viewModel.okulu = person.militarySchool;
            that.viewModel.sicilNo = person.registryNumber;
            that.viewModel.nasbi = person.lastNominationDate;
            that.viewModel.kayitliOlduguAskerlikSubesi = person.militaryOffice;
            that.viewModel.ucucuGorevi = person.flyingMission;
            that.viewModel.uctuguUcakTipi = person.flyingPlaneType;

            if (person.hcAdmissionTypeDef == Guid.empty().valueOf().toString()) {
              that.viewModel.skBasvuruGrubu = null;
            } else {
              that.viewModel.skBasvuruGrubu = person.hcAdmissionTypeDef;
            }


            if (personRes.townOfBirthCode) {
              that.viewModel.dogumYeri = personRes.townOfBirthCode.toString();
            }
            if (personRes.nationalityCode) {
              that.viewModel.uyrugu = personRes.nationalityCode;
            }
          } 
         
          else {
            that.messageService.showError(personRes.sonucKodu + " - " + personRes.sonucMesaji);

            //Eğer Model Dolu ise boşaltsın
             this.viewModel = new SevkYeniGirisFoto();
            //this.messageService.showError(`${that.uniqueRefno} - T.C. kimlik numarası ile kişi kaydı bulunamadı.`);
          }
        });
      }
      this.personGetInput.uniqueRefno = undefined;
    }
    if (this.isTcOrForeign === this.identificationnumbers[1]) {
      if (that.foreignUniqueRefno === null || that.foreignUniqueRefno === undefined) {
        this.messageService.showWarning(`Geçerli yabancı kimlik numarası girmelisiniz!`);
       // confirm(`Geçerli yabancı kimlik numarası girmelisiniz!`);
        that.foreignUniqueRefno = undefined;
      }
      if (that.foreignUniqueRefno !== null && that.foreignUniqueRefno !== undefined) {
        this.personGetInput.foreignUniqueRefno = that.foreignUniqueRefno;
        this.personService.getPerson(this.personGetInput).toPromise().then((personRes: PersonResult) => {
          if (personRes && personRes.person) {
            const person = personRes.person;
            that.viewModel.yabanciKimlikNo = person.foreignUniqueRefno;
            that.viewModel.adi = person.name;
            that.viewModel.soyadi = person.surname;
            that.viewModel.anneAdi = person.motherName;
            that.viewModel.babaAdi = person.fatherName;
            that.viewModel.cinsiyet = person.gender;
            that.viewModel.dogumTarihi = person.birthDate;
            that.viewModel.adres = person.address;
            that.viewModel.skBasvuruGrubu = person.hcAdmissionTypeDef;
            that.viewModel.kuvvet = person.forcesCommand;
            that.viewModel.sinif = person.militaryClass;
            that.viewModel.rutbe = person.rank;
            that.viewModel.birlik = person.militaryUnit;
            that.viewModel.birlikAciklama = person.militaryUnitDescription;
            that.viewModel.okulu = person.militarySchool;
            that.viewModel.sicilNo = person.registryNumber;
            that.viewModel.nasbi = person.lastNominationDate;
            that.viewModel.kayitliOlduguAskerlikSubesi = person.militaryOffice;
            that.viewModel.ucucuGorevi = person.flyingMission;
            that.viewModel.uctuguUcakTipi = person.flyingPlaneType;
            if (personRes.townOfBirthCode) {
              that.viewModel.dogumYeri = personRes.townOfBirthCode.toString();
            }
            if (personRes.nationalityCode) {
              that.viewModel.uyrugu = personRes.nationalityCode;
            }
          } else {
            this.messageService.showError(`Yabancı kimlik numarası ile kişi kaydı bulunamadı.`);
          }
        });
      }
      this.personGetInput.foreignUniqueRefno = undefined;
    }
  }

  onSevkKapsamiChanged(data) {
    if (data.value !== undefined) {
      if (data.value === 1) {
        this.showSevkKapsamiSehir = true;
        this.showSevkKapsamiHastane = true;
      } if (data.value === 2) {
        this.showSevkKapsamiSehir = false;
        this.showSevkKapsamiHastane = true;
      } if (data.value === 3) {
        this.showSevkKapsamiSehir = true;
        this.showSevkKapsamiHastane = false;
      }
    }
  }

  public ngAfterViewInit(): void {
    const that = this;
    if (this.form) {
      setTimeout(function () {
        that.validateAllFields(that.form.control);
      }, 10);
    }
  }

  private validateAllFields(formGroup: FormGroup) {
    const keys = Object.keys(formGroup.controls);
    for (const key of keys) {
      const control = formGroup.get(key);
      if (control instanceof FormControl) {
        control.markAsTouched({ onlySelf: true });
      } else if (control instanceof FormGroup) {
        this.validateAllFields(control);
      }
    }
  }

      this.showTcNoInput = false;

    if ($event.value && $event.value === this.identificationnumbers[1]) {
      this.showTcNoInput = true;
      this.showForeignNoInput = false;
      this.uniqueRefno = undefined;
    }
  }

  public send2qrCode() {
    this.closeCam();
    const that = this;
    //yeni qr okutuldugunda silme işlemi yapılması için eklendi.
    if (this.imageUploader) {
      this.imageUploader.removeImage();
    }
    if (this.viewModel.TempBinaryDataObjectId)
      this.viewModel.TempBinaryDataObjectId = null;

    if (this.qrcode == null || this.qrcode == "") {
      this.messageService.showError('QR kod Boş olamaz');
      return;
    }
    if(!this.qrcode.includes(',')){
      this.messageService.showError('QR kodu hatalı');
    }
    else {
      let re = /\"/gi;
      this.qrcode = this.qrcode.replace(re, '');
      const qrCodeConvert2Json = this.qrcode.split(',').map(keyVal => {
        return keyVal
          .split(':')
          .map(_ => _.trim())
      })
        .reduce((accumulator, currentValue) => {
          accumulator[currentValue[0]] = currentValue[1]
          return accumulator
        }, {});

        if(qrCodeConvert2Json["telefonNo"].length != 10){
          this.messageService.showError(`Telefon Numarası 10 haneli olmalı`);
          return
        }
      //for (let item in qrCodeConvert2Json) {
        this.personGetInput.uniqueRefno = qrCodeConvert2Json["tcKimlikNo"];

private previewFile(
    responseBody: any,
    contentType: string,
    fileName: string
) {
    const blob = new Blob(
        [responseBody],
        { type: contentType }
    );

    const url = window.URL.createObjectURL(blob);

    window.open(url, '_blank');
}

public dispatchFileSelectionChanged(e: any) {

    if (e.rowType !== 'data') {
        return;
    }

    const dispatchFileId = e.data.dispatchFileId;

    if (!dispatchFileId) {
        return;
    }

    console.log('Tıklanan dosya:', dispatchFileId);

    this.fileUploadService
        .downloadDispatchFile(dispatchFileId)
        .toPromise()
        .then(response => {

            const contentType =
                response.headers.get('content-type') ||
                'application/octet-stream';

            console.log('Gelen dosya tipi:', contentType);

            this.previewFile(
                response.body,
                contentType,
                dispatchFileId
            );

        })
        .catch(error => {
            console.error('Dosya getirilemedi:', error);
        });
}
