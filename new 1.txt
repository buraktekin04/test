namespace MSKBS.Domain.Common;

/// <summary>
/// Sistemde kalıcı olarak saklanan entity'lerin ortak temel sınıfıdır.
/// Kimlik, audit, aktiflik, soft delete ve optimistic concurrency
/// bilgilerini merkezi olarak taşır.
/// </summary>
public abstract class BaseEntity
{
    /// <summary>
    /// Kaydın sistem genelindeki benzersiz kimliğidir.
    /// </summary>
    public Guid Id { get; set; } = Guid.NewGuid();

    /// <summary>
    /// Kaydın oluşturulduğu UTC tarih ve saattir.
    /// AuditableEntitySaveChangesInterceptor tarafından otomatik doldurulur.
    /// </summary>
    public DateTime CreatedDate { get; set; }

    /// <summary>
    /// Kaydı oluşturan kullanıcının kimliğidir.
    /// Sistem tarafından oluşturulan kayıtlarda boş olabilir.
    /// </summary>
    public Guid? CreatedBy { get; set; }

    /// <summary>
    /// Kaydın son güncellendiği UTC tarih ve saattir.
    /// </summary>
    public DateTime? UpdatedDate { get; set; }

    /// <summary>
    /// Kaydı son güncelleyen kullanıcının kimliğidir.
    /// </summary>
    public Guid? UpdatedBy { get; set; }

    /// <summary>
    /// Kaydın iş süreçlerinde kullanılabilir durumda olup olmadığını belirtir.
    /// Pasif kayıt silinmiş kayıt anlamına gelmez.
    /// </summary>
    public bool IsActive { get; set; } = true;

    /// <summary>
    /// Kaydın soft delete yöntemiyle silinmiş olarak işaretlenip
    /// işaretlenmediğini belirtir.
    /// </summary>
    public bool IsDeleted { get; set; } = false;

    /// <summary>
    /// Kaydın soft delete işlemine tabi tutulduğu UTC tarih ve saattir.
    /// </summary>
    public DateTime? DeletedDate { get; set; }

    /// <summary>
    /// Kaydı soft delete yöntemiyle silen kullanıcının kimliğidir.
    /// </summary>
    public Guid? DeletedBy { get; set; }

    /// <summary>
    /// Optimistic concurrency kontrolünde kullanılan satır versiyonudur.
    /// PostgreSQL tarafında fiziksel bir Version kolonu oluşturulmaz;
    /// Npgsql tarafından PostgreSQL'in xmin sistem kolonuna map edilir.
    /// </summary>
    public uint Version { get; set; }
}

using MSKBS.Domain.Common;
using MSKBS.Domain.Entities.Organization;

namespace MSKBS.Domain.Entities.Identity;

/// <summary>
/// Sistemde kimlik doğrulaması yapılabilen kullanıcı hesabını temsil eder.
/// Kullanıcının kimlik, güvenlik, organizasyon ve yetkilendirme
/// ilişkilerini üzerinde taşır.
/// </summary>
public sealed class User : BaseEntity
{
    /// <summary>
    /// Kullanıcının sisteme giriş yapmak için kullandığı kullanıcı adıdır.
    /// </summary>
    public string UserName { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcı adının karşılaştırma ve benzersizlik kontrollerinde
    /// kullanılan normalize edilmiş halidir.
    /// </summary>
    public string NormalizedUserName { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcının güvenli şekilde hash'lenmiş parolasıdır.
    /// Açık parola hiçbir zaman veritabanında tutulmaz.
    /// </summary>
    public string PasswordHash { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcının adıdır.
    /// </summary>
    public string FirstName { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcının soyadıdır.
    /// </summary>
    public string LastName { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcının zorunlu e-posta adresidir.
    /// </summary>
    public string Email { get; set; } = string.Empty;

    /// <summary>
    /// E-posta adresinin karşılaştırma ve benzersizlik kontrollerinde
    /// kullanılan normalize edilmiş halidir.
    /// </summary>
    public string NormalizedEmail { get; set; } = string.Empty;

    /// <summary>
    /// Kullanıcının telefon numarasıdır.
    /// </summary>
    public string? PhoneNumber { get; set; }

    /// <summary>
    /// Kullanıcının bağlı olduğu organizasyon biriminin kimliğidir.
    /// Sistem veya teknik kullanıcılarda boş olabilir.
    /// </summary>
    public Guid? OrganizationUnitId { get; set; }

    /// <summary>
    /// Kullanıcının bağlı olduğu organizasyon birimini temsil eder.
    /// </summary>
    public OrganizationUnit? OrganizationUnit { get; set; }

    /// <summary>
    /// Art arda gerçekleşen başarısız giriş denemelerinin sayısını tutar.
    /// </summary>
    public int AccessFailedCount { get; set; }

    /// <summary>
    /// Kullanıcı hesabının kilitlenmiş durumda olup olmadığını belirtir.
    /// </summary>
    public bool IsLocked { get; set; }

    /// <summary>
    /// Hesap kilidinin sona ereceği UTC tarih ve saattir.
    /// </summary>
    public DateTime? LockoutEnd { get; set; }

    /// <summary>
    /// Kullanıcı için iki faktörlü kimlik doğrulamanın
    /// etkin olup olmadığını belirtir.
    /// </summary>
    public bool TwoFactorAuthenticationEnabled { get; set; }

    /// <summary>
    /// Kullanıcı hesabının geçerli olmaya başladığı UTC tarih ve saattir.
    /// </summary>
    public DateTime? ValidFrom { get; set; }

    /// <summary>
    /// Kullanıcı hesabının geçerliliğinin sona ereceği UTC tarih ve saattir.
    /// Süresiz hesaplarda boş olabilir.
    /// </summary>
    public DateTime? ValidTo { get; set; }

    /// <summary>
    /// Kullanıcının sisteme son başarılı giriş yaptığı UTC tarih ve saattir.
    /// </summary>
    public DateTime? LastLoginDate { get; set; }

    /// <summary>
    /// Kullanıcı parolasının en son değiştirildiği UTC tarih ve saattir.
    /// </summary>
    public DateTime? PasswordChangedDate { get; set; }

    /// <summary>
    /// Kullanıcının güvenlik durumunu temsil eden rastgele değerdir.
    /// Parola veya kritik güvenlik bilgileri değiştirildiğinde yenilenebilir.
    /// </summary>
    public string SecurityStamp { get; set; } = Guid.NewGuid().ToString("N");

    /// <summary>
    /// Kullanıcıya atanmış rol ilişkilerini temsil eder.
    /// </summary>
    public ICollection<UserRole> UserRoles { get; set; }
        = new List<UserRole>();

    /// <summary>
    /// Kullanıcıya özel permission override kayıtlarını temsil eder.
    /// </summary>
    public ICollection<UserPermission> UserPermissions { get; set; }
        = new List<UserPermission>();

    /// <summary>
    /// Kullanıcıya ait refresh token kayıtlarını temsil eder.
    /// </summary>
    public ICollection<RefreshToken> RefreshTokens { get; set; }
        = new List<RefreshToken>();

    /// <summary>
    /// Kullanıcı için oluşturulmuş parola sıfırlama token kayıtlarını temsil eder.
    /// </summary>
    public ICollection<PasswordResetToken> PasswordResetTokens { get; set; }
        = new List<PasswordResetToken>();
}

10. RefreshToken.cs
using MSKBS.Domain.Common;

namespace MSKBS.Domain.Entities.Identity;

/// <summary>
/// JWT access token yenileme işlemlerinde kullanılan refresh token kaydını
/// temsil eder.
/// Açık token veritabanında tutulmaz; yalnızca hash değeri saklanır.
/// </summary>
public sealed class RefreshToken : BaseEntity
{
    /// <summary>
    /// Refresh token'ın ait olduğu kullanıcının kimliğidir.
    /// </summary>
    public Guid UserId { get; set; }

    /// <summary>
    /// Refresh token'ın güvenli hash değeridir.
    /// </summary>
    public string TokenHash { get; set; } = string.Empty;

    /// <summary>
    /// Refresh token'ın geçerliliğinin sona ereceği UTC tarih ve saattir.
    /// </summary>
    public DateTime ExpiresAt { get; set; }

    /// <summary>
    /// Token revoke edilmişse işlemin gerçekleştiği UTC tarih ve saattir.
    /// </summary>
    public DateTime? RevokedAt { get; set; }

    /// <summary>
    /// Token rotation sonucunda bu token'ın yerine oluşturulan
    /// yeni refresh token kaydının kimliğidir.
    /// </summary>
    public Guid? ReplacedByTokenId { get; set; }

    /// <summary>
    /// Token oluşturulurken isteği yapan istemcinin IP adresidir.
    /// </summary>
    public string? CreatedByIp { get; set; }

    /// <summary>
    /// Token revoke edilirken isteği yapan istemcinin IP adresidir.
    /// </summary>
    public string? RevokedByIp { get; set; }

    /// <summary>
    /// Token'ın revoke edilme nedenidir.
    /// </summary>
    public string? RevocationReason { get; set; }

    /// <summary>
    /// Token oluşturulurken kullanılan istemcinin User-Agent bilgisidir.
    /// </summary>
    public string? UserAgent { get; set; }

    /// <summary>
    /// Kullanıcıya gösterilmek üzere istemcinin veya cihazın okunabilir adıdır.
    /// </summary>
    public string? DeviceName { get; set; }

    /// <summary>
    /// Refresh token'ın ait olduğu kullanıcıyı temsil eder.
    /// </summary>
    public User User { get; set; } = null!;

    /// <summary>
    /// Token rotation sonucunda bu token'ın yerine oluşturulan
    /// refresh token kaydını temsil eder.
    /// </summary>
    public RefreshToken? ReplacedByToken { get; set; }

    /// <summary>
    /// Token'ın geçerlilik süresinin dolup dolmadığını belirtir.
    /// Veritabanında saklanmaz.
    /// </summary>
    public bool IsExpired => DateTime.UtcNow >= ExpiresAt;

    /// <summary>
    /// Token'ın revoke edilip edilmediğini belirtir.
    /// Veritabanında saklanmaz.
    /// </summary>
    public bool IsRevoked => RevokedAt.HasValue;

    /// <summary>
    /// Refresh token'ın halen kullanılabilir olup olmadığını belirtir.
    /// Veritabanında saklanmaz.
    /// </summary>
    public bool IsValid =>
        IsActive &&
        !IsDeleted &&
        !IsRevoked &&
        !IsExpired;
}



11. PasswordResetToken.cs
using MSKBS.Domain.Common;

namespace MSKBS.Domain.Entities.Identity;

/// <summary>
/// Kullanıcının parola sıfırlama işlemi için oluşturulan tek kullanımlık
/// güvenlik token'ını temsil eder.
/// Açık token veritabanında tutulmaz; yalnızca hash değeri saklanır.
/// </summary>
public sealed class PasswordResetToken : BaseEntity
{
    /// <summary>
    /// Parola sıfırlama token'ının ait olduğu kullanıcının kimliğidir.
    /// </summary>
    public Guid UserId { get; set; }

    /// <summary>
    /// Parola sıfırlama token'ının güvenli hash değeridir.
    /// </summary>
    public string TokenHash { get; set; } = string.Empty;

    /// <summary>
    /// Token'ın geçerliliğinin sona ereceği UTC tarih ve saattir.
    /// </summary>
    public DateTime ExpiresAt { get; set; }

    /// <summary>
    /// Token başarılı bir parola sıfırlama işleminde kullanılmışsa
    /// kullanıldığı UTC tarih ve saattir.
    /// </summary>
    public DateTime? UsedAt { get; set; }

    /// <summary>
    /// Token kullanılamaz hale getirilmişse revoke edildiği
    /// UTC tarih ve saattir.
    /// </summary>
    public DateTime? RevokedAt { get; set; }

    /// <summary>
    /// Token'ın ait olduğu kullanıcıyı temsil eder.
    /// </summary>
    public User User { get; set; } = null!;

    /// <summary>
    /// Token'ın geçerlilik süresinin dolup dolmadığını belirtir.
    /// </summary>
    public bool IsExpired => DateTime.UtcNow >= ExpiresAt;

    /// <summary>
    /// Token'ın daha önce kullanılıp kullanılmadığını belirtir.
    /// </summary>
    public bool IsUsed => UsedAt.HasValue;

    /// <summary>
    /// Token'ın revoke edilip edilmediğini belirtir.
    /// </summary>
    public bool IsRevoked => RevokedAt.HasValue;

    /// <summary>
    /// Token'ın halen parola sıfırlama işleminde kullanılabilir
    /// durumda olup olmadığını belirtir.
    /// </summary>
    public bool IsValid =>
        IsActive &&
        !IsDeleted &&
        !IsExpired &&
        !IsUsed &&
        !IsRevoked;
}

/// <summary>

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using MSKBS.Domain.Common;

namespace MSKBS.Persistence.Configurations.Common;

/// <summary>
/// BaseEntity'den türeyen entity'ler için ortak EF Core
/// veritabanı yapılandırmalarını tanımlar.
/// </summary>
/// <typeparam name="TEntity">
/// Yapılandırılacak BaseEntity türevi entity tipidir.
/// </typeparam>
public abstract class BaseEntityConfiguration<TEntity>
    : IEntityTypeConfiguration<TEntity>
    where TEntity : BaseEntity
{
    /// <summary>
    /// BaseEntity ortak mapping kurallarını ve ardından
    /// entity'ye özel configuration kurallarını uygular.
    /// </summary>
    /// <param name="builder">
    /// İlgili entity için EF Core builder nesnesidir.
    /// </param>
    public void Configure(EntityTypeBuilder<TEntity> builder)
    {
        ConfigureBaseEntity(builder);
        ConfigureEntity(builder);
    }

    /// <summary>
    /// BaseEntity üzerinden gelen ortak property'lerin
    /// veritabanı mapping kurallarını uygular.
    /// </summary>
    /// <param name="builder">
    /// İlgili entity için EF Core builder nesnesidir.
    /// </param>
    protected virtual void ConfigureBaseEntity(
        EntityTypeBuilder<TEntity> builder)
    {
        builder.HasKey(entity => entity.Id);

        builder.Property(entity => entity.Id)
            .ValueGeneratedNever();

        builder.Property(entity => entity.CreatedDate)
            .IsRequired()
            .HasColumnType("timestamp with time zone");

        builder.Property(entity => entity.CreatedBy);

        builder.Property(entity => entity.UpdatedDate)
            .HasColumnType("timestamp with time zone");

        builder.Property(entity => entity.UpdatedBy);

        builder.Property(entity => entity.IsActive)
            .IsRequired()
            .HasDefaultValue(true);

        builder.Property(entity => entity.IsDeleted)
            .IsRequired()
            .HasDefaultValue(false);

        builder.Property(entity => entity.DeletedDate)
            .HasColumnType("timestamp with time zone");

        builder.Property(entity => entity.DeletedBy);

        /*
         * Npgsql, uint + IsRowVersion() kombinasyonunu PostgreSQL'in
         * xmin sistem kolonu ile optimistic concurrency kontrolünde kullanır.
         *
         * Migration sırasında ayrıca Version isimli fiziksel kolon
         * oluşturulması beklenmez.
         */
        builder.Property(entity => entity.Version)
            .IsRowVersion();
    }

    /// <summary>
    /// Türetilen configuration sınıfının entity'ye özel mapping
    /// kurallarını tanımladığı metottur.
    /// </summary>
    /// <param name="builder">
    /// İlgili entity için EF Core builder nesnesidir.
    /// </param>
    protected abstract void ConfigureEntity(
        EntityTypeBuilder<TEntity> builder);
}

19. ModelBuilderExtensions.cs
MSKBS.Persistence/Extensions/ModelBuilderExtensions.cs
Global query filter'ların tek merkezi burası.

using System.Linq.Expressions;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata;
using MSKBS.Domain.Common;
using MSKBS.Persistence.Configurations.Common;

namespace MSKBS.Persistence.Extensions;

/// <summary>
/// EF Core model oluşturma sürecinde kullanılan ortak
/// model yapılandırma extension metotlarını içerir.
/// </summary>
public static class ModelBuilderExtensions
{
    /// <summary>
    /// BaseEntity'den türeyen bütün root entity tiplerine
    /// SoftDeleteFilter ve ActiveFilter named global query
    /// filter'larını uygular.
    /// </summary>
    /// <param name="modelBuilder">
    /// EF Core model yapılandırmasını temsil eden ModelBuilder nesnesidir.
    /// </param>
    public static void ApplyGlobalQueryFilters(
        this ModelBuilder modelBuilder)
    {
        var entityTypes = modelBuilder.Model
            .GetEntityTypes()
            .Where(IsBaseEntityRootType)
            .ToList();

        foreach (var entityType in entityTypes)
        {
            var softDeleteFilter = CreateBooleanFilter(
                entityType.ClrType,
                nameof(BaseEntity.IsDeleted),
                expectedValue: false);

            var activeFilter = CreateBooleanFilter(
                entityType.ClrType,
                nameof(BaseEntity.IsActive),
                expectedValue: true);

            modelBuilder.Entity(entityType.ClrType)
                .HasQueryFilter(
                    QueryFilterNames.SoftDelete,
                    softDeleteFilter);

            modelBuilder.Entity(entityType.ClrType)
                .HasQueryFilter(
                    QueryFilterNames.Active,
                    activeFilter);
        }
    }

    /// <summary>
    /// Verilen entity tipinin BaseEntity'den türeyen ve inheritance
    /// hiyerarşisinin root entity tipi olup olmadığını belirler.
    /// </summary>
    /// <param name="entityType">
    /// Kontrol edilecek EF Core entity metadata nesnesidir.
    /// </param>
    /// <returns>
    /// Entity global query filter uygulanabilecek bir BaseEntity root tipi ise
    /// true; aksi durumda false döner.
    /// </returns>
    private static bool IsBaseEntityRootType(
        IReadOnlyEntityType entityType)
    {
        return typeof(BaseEntity)
                   .IsAssignableFrom(entityType.ClrType)
               && entityType.BaseType is null;
    }

    /// <summary>
    /// Belirtilen boolean property için dinamik global query
    /// filter expression'ı oluşturur.
    /// </summary>
    /// <param name="entityType">
    /// Filtrenin uygulanacağı CLR entity tipidir.
    /// </param>
    /// <param name="propertyName">
    /// Filtre uygulanacak boolean property'nin adıdır.
    /// </param>
    /// <param name="expectedValue">
    /// Property'nin sahip olması beklenen değerdir.
    /// </param>
    /// <returns>
    /// EF Core tarafından query filter olarak kullanılabilecek
    /// lambda expression döner.
    /// </returns>
    private static LambdaExpression CreateBooleanFilter(
        Type entityType,
        string propertyName,
        bool expectedValue)
    {
        var parameter = Expression.Parameter(
            entityType,
            "entity");

        var property = Expression.Property(
            parameter,
            propertyName);

        var expectedValueExpression =
            Expression.Constant(expectedValue);

        var body = Expression.Equal(
            property,
            expectedValueExpression);

        return Expression.Lambda(
            body,
            parameter);
    }
}

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using MSKBS.Domain.Entities.Identity;
using MSKBS.Persistence.Configurations.Common;

namespace MSKBS.Persistence.Configurations.Identity;

/// <summary>
/// RefreshToken entity'sinin PostgreSQL ve EF Core yapılandırmasını tanımlar.
/// Refresh token güvenliği, rotation zinciri ve kullanıcı ilişkisini yönetir.
/// </summary>
public sealed class RefreshTokenConfiguration
    : BaseEntityConfiguration<RefreshToken>
{
    /// <summary>
    /// RefreshToken entity'sine özgü kolon, index, foreign key ve
    /// token rotation ilişkilerini yapılandırır.
    /// </summary>
    /// <param name="builder">
    /// RefreshToken entity'si için EF Core builder nesnesidir.
    /// </param>
    protected override void ConfigureEntity(
        EntityTypeBuilder<RefreshToken> builder)
    {
        builder.ToTable("RefreshTokens");

        builder.Property(refreshToken => refreshToken.TokenHash)
            .IsRequired()
            .HasMaxLength(128);

        builder.Property(refreshToken => refreshToken.ExpiresAt)
            .IsRequired()
            .HasColumnType("timestamp with time zone");

        builder.Property(refreshToken => refreshToken.RevokedAt)
            .HasColumnType("timestamp with time zone");

        builder.Property(refreshToken => refreshToken.CreatedByIp)
            .HasMaxLength(64);

        builder.Property(refreshToken => refreshToken.RevokedByIp)
            .HasMaxLength(64);

        builder.Property(refreshToken => refreshToken.RevocationReason)
            .HasMaxLength(500);

        builder.Property(refreshToken => refreshToken.UserAgent)
            .HasMaxLength(1024);

        builder.Property(refreshToken => refreshToken.DeviceName)
            .HasMaxLength(200);

        builder.HasOne(refreshToken => refreshToken.User)
            .WithMany(user => user.RefreshTokens)
            .HasForeignKey(refreshToken => refreshToken.UserId)
            .OnDelete(DeleteBehavior.Restrict);

        /*
         * Token rotation zinciri:
         *
         * Token A
         *    ↓
         * Token B
         *    ↓
         * Token C
         *
         * Eski token silindiğinde veya kullanıcı silindiğinde rotation
         * geçmişinin otomatik olarak fiziksel silinmesini istemiyoruz.
         */
        builder.HasOne(refreshToken => refreshToken.ReplacedByToken)
            .WithMany()
            .HasForeignKey(refreshToken => refreshToken.ReplacedByTokenId)
            .OnDelete(DeleteBehavior.Restrict);

        /*
         * TokenHash geçmiş kayıtlar dahil sistem genelinde benzersizdir.
         *
         * Burada partial unique index kullanmıyoruz.
         * Silinmiş/revoke edilmiş bir token'ın hash değerinin dahi
         * başka bir token tarafından tekrar kullanılmasını istemiyoruz.
         */
        builder.HasIndex(refreshToken => refreshToken.TokenHash)
            .IsUnique();

        /*
         * Kullanıcıya ait token/session kayıtlarının hızlı sorgulanması
         * için index oluşturulur.
         */
        builder.HasIndex(refreshToken => refreshToken.UserId);

        /*
         * Süresi dolmuş token kayıtlarının temizlik işlemlerinde
         * hızlı bulunmasını sağlar.
         */
        builder.HasIndex(refreshToken => refreshToken.ExpiresAt);

        /*
         * Bir refresh token yalnızca tek bir önceki token'ın replacement'ı
         * olabilir. Böylece rotation zincirinin dallanması engellenir.
         *
         * PostgreSQL unique index birden fazla NULL değere izin verdiğinden
         * ayrıca partial filter yazmaya gerek yoktur.
         */
        builder.HasIndex(
                refreshToken => refreshToken.ReplacedByTokenId)
            .IsUnique();

        /*
         * Hesaplanan domain property'leri fiziksel kolon değildir.
         */
        builder.Ignore(refreshToken => refreshToken.IsExpired);
        builder.Ignore(refreshToken => refreshToken.IsRevoked);
        builder.Ignore(refreshToken => refreshToken.IsValid);
    }
}

                passwordResetToken.User)
            .WithMany(user => user.PasswordResetTokens)
            .HasForeignKey(passwordResetToken =>
                passwordResetToken.UserId)
            .OnDelete(DeleteBehavior.Restrict);

        /*
         * Reset token hash'i geçmiş kayıtlar dahil benzersizdir.
         */
        builder.HasIndex(passwordResetToken =>
                passwordResetToken.TokenHash)
            .IsUnique();

        /*
         * Kullanıcının aktif veya geçmiş parola sıfırlama token'larını
         * sorgulamak için kullanılır.
         */
        builder.HasIndex(passwordResetToken =>
            passwordResetToken.UserId);

        /*
         * Süresi dolmuş token kayıtlarının bulunması ve temizlenmesi
         * için kullanılır.
         */
        builder.HasIndex(passwordResetToken =>
            passwordResetToken.ExpiresAt);

        builder.Ignore(passwordResetToken =>
            passwordResetToken.IsExpired);

        builder.Ignore(passwordResetToken =>
            passwordResetToken.IsUsed);

        builder.Ignore(passwordResetToken =>
            passwordResetToken.IsRevoked);

        builder.Ignore(passwordResetToken =>
            passwordResetToken.IsValid);
    }
}

/// <summary>
/// OrganizationUnit entity'sinin PostgreSQL ve EF Core
/// veritabanı yapılandırmasını tanımlar.
/// Organizasyon ağacının parent-child ilişkisini yönetir.
/// </summary>
public sealed class OrganizationUnitConfiguration
    : BaseEntityConfiguration<OrganizationUnit>
{
    /// <summary>
    /// OrganizationUnit entity'sine özgü kolon, self-reference ilişki,
    /// foreign key ve index kurallarını uygular.
    /// </summary>
    /// <param name="builder">
    /// OrganizationUnit entity'si için EF Core builder nesnesidir.
    /// </param>
    protected override void ConfigureEntity(
        EntityTypeBuilder<OrganizationUnit> builder)
    {
        builder.ToTable("OrganizationUnits");

        builder.Property(organizationUnit =>
                organizationUnit.Code)
            .IsRequired()
            .HasMaxLength(100);

        builder.Property(organizationUnit =>
                organizationUnit.Name)
            .IsRequired()
            .HasMaxLength(250);

        /*
         * Parent-child organizasyon ilişkisi.
         *
         * Parent silindiğinde child birimlerin fiziksel olarak
         * silinmesini istemiyoruz.
         *
         * Zaten sistemde soft delete kullanılacağı için fiziksel
         * cascade delete doğru davranış değildir.
         */
        builder.HasOne(organizationUnit =>
                organizationUnit.ParentOrganizationUnit)
            .WithMany(organizationUnit =>
                organizationUnit.Children)
            .HasForeignKey(organizationUnit =>
                organizationUnit.ParentOrganizationUnitId)
            .OnDelete(DeleteBehavior.Restrict);

        /*
         * OrganizationUnit.Code teknik referans olarak kullanılacağından
         * silinmemiş kayıtlar arasında benzersizdir.
         *
         * Soft delete edilmiş bir organizasyon kodu daha sonra yeniden
         * kullanılabilir.
         */
        builder.HasIndex(organizationUnit =>
                organizationUnit.Code)
            .IsUnique()
            .HasFilter(PostgreSqlIndexFilters.NotDeleted);

        /*
         * Parent üzerinden child birimlerin hızlı listelenmesini sağlar.
         */
        builder.HasIndex(organizationUnit =>
            organizationUnit.ParentOrganizationUnitId);

        /*
         * Organizasyon listeleme ve temel isim sorgularında yardımcı olur.
         * Name unique değildir; farklı organizasyon dallarında aynı isimli
         * birimler bulunabilir.
         */
        builder.HasIndex(organizationUnit =>
            organizationUnit.Name);
    }

public sealed class CurrentUserService : ICurrentUserService
{

    /// <summary>
    /// CurrentUserService sınıfının yeni bir örneğini oluşturur.
    /// </summary>

            .IsAuthenticated == true;

        _httpContextAccessor
            .HttpContext?
            .Connection
            .RemoteIpAddress?

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Diagnostics;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Domain.Common;

namespace MSKBS.Persistence.Interceptors;

/// <summary>
/// BaseEntity'den türeyen kayıtların oluşturulma, güncellenme ve
/// soft delete audit alanlarını SaveChanges sürecinde merkezi olarak yönetir.
///
/// Handler'ların CreatedDate, UpdatedDate, DeletedDate veya kullanıcı
/// audit alanlarını elle doldurmasına ihtiyaç bırakmaz.
/// </summary>
public sealed class AuditableEntitySaveChangesInterceptor
    : SaveChangesInterceptor
{
    private readonly ICurrentUserService _currentUserService;

    /// <summary>
    /// AuditableEntitySaveChangesInterceptor sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="currentUserService">
    /// İşlemi gerçekleştiren mevcut kullanıcıya ait bilgileri sağlayan servistir.
    /// </param>
    public AuditableEntitySaveChangesInterceptor(
        ICurrentUserService currentUserService)
    {
        _currentUserService = currentUserService;
    }

    /// <summary>
    /// Senkron SaveChanges çağrısından önce audit ve soft delete
    /// alanlarını günceller.
    /// </summary>
    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData,
        InterceptionResult<int> result)
    {
        if (eventData.Context is not null)
        {
            ApplyAuditInformation(eventData.Context);
        }

        return base.SavingChanges(eventData, result);
    }

    /// <summary>
    /// Asenkron SaveChanges çağrısından önce audit ve soft delete
    /// alanlarını günceller.
    /// </summary>
    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData,
        InterceptionResult<int> result,
        CancellationToken cancellationToken = default)
    {
        if (eventData.Context is not null)
        {
            ApplyAuditInformation(eventData.Context);
        }

        return base.SavingChangesAsync(
            eventData,
            result,
            cancellationToken);
    }

    /// <summary>
    /// ChangeTracker içerisindeki BaseEntity kayıtlarını inceleyerek
    /// oluşturma, güncelleme ve soft delete audit alanlarını uygular.
    /// </summary>
    /// <param name="dbContext">
    /// Değişikliklerin takip edildiği EF Core DbContext nesnesidir.
    /// </param>
    private void ApplyAuditInformation(DbContext dbContext)
    {
        var utcNow = DateTime.UtcNow;
        var currentUserId = _currentUserService.UserId;

        var entries = dbContext.ChangeTracker
            .Entries<BaseEntity>()
            .Where(entry =>
                entry.State is EntityState.Added
                    or EntityState.Modified
                    or EntityState.Deleted)
            .ToList();

        foreach (var entry in entries)
        {
            switch (entry.State)
            {
                case EntityState.Added:
                    ApplyCreatedAudit(
                        entry.Entity,
                        utcNow,
                        currentUserId);
                    break;

                case EntityState.Modified:
                    ApplyUpdatedAudit(
                        entry.Entity,
                        utcNow,
                        currentUserId);
                    break;

                case EntityState.Deleted:
                    ApplySoftDelete(
                        entry,
                        utcNow,
                        currentUserId);
                    break;
            }
        }
    }

    /// <summary>
    /// Yeni oluşturulan entity'nin oluşturulma audit alanlarını doldurur.
    /// </summary>
    private static void ApplyCreatedAudit(
        BaseEntity entity,
        DateTime utcNow,
        Guid? currentUserId)
    {
        entity.CreatedDate = utcNow;
        entity.CreatedBy = currentUserId;

        entity.IsActive = true;
        entity.IsDeleted = false;

        entity.UpdatedDate = null;
        entity.UpdatedBy = null;
        entity.DeletedDate = null;
        entity.DeletedBy = null;
    }

    /// <summary>
    /// Güncellenen entity'nin güncelleme audit alanlarını doldurur.
    /// </summary>
    private static void ApplyUpdatedAudit(
        BaseEntity entity,
        DateTime utcNow,
        Guid? currentUserId)
    {
        entity.UpdatedDate = utcNow;
        entity.UpdatedBy = currentUserId;
    }

    /// <summary>
    /// Fiziksel delete işlemini soft delete işlemine dönüştürür ve
    /// silme audit alanlarını doldurur.
    /// </summary>
    private static void ApplySoftDelete(
        Microsoft.EntityFrameworkCore.ChangeTracking.EntityEntry<BaseEntity> entry,
        DateTime utcNow,
        Guid? currentUserId)
    {
        entry.State = EntityState.Modified;

        entry.Entity.IsDeleted = true;
        entry.Entity.IsActive = false;

        entry.Entity.DeletedDate = utcNow;
        entry.Entity.DeletedBy = currentUserId;

        entry.Entity.UpdatedDate = utcNow;
        entry.Entity.UpdatedBy = currentUserId;
    }
}

3. CurrentUserService güncel hali
MSKBS.Infrastructure/Identity/CurrentUserService.cs

using System.Security.Claims;
using Microsoft.AspNetCore.Http;
using MSKBS.Application.Common.Constants;
using MSKBS.Application.Common.Interfaces.Identity;

namespace MSKBS.Infrastructure.Identity;

/// <summary>
/// Mevcut HTTP isteğindeki ClaimsPrincipal üzerinden kullanıcıya ait
/// kimlik, organizasyon, rol, permission ve istemci bilgilerini sağlar.
/// </summary>
public sealed class CurrentUserService : ICurrentUserService
{
    private readonly IHttpContextAccessor _httpContextAccessor;

    /// <summary>
    /// CurrentUserService sınıfının yeni bir örneğini oluşturur.
    /// </summary>
    /// <param name="httpContextAccessor">
    /// Mevcut HTTP context'ine erişim sağlayan servistir.
    /// </param>
    public CurrentUserService(
        IHttpContextAccessor httpContextAccessor)
    {
        _httpContextAccessor = httpContextAccessor;
    }

    /// <summary>
    /// Mevcut HTTP isteğinin ClaimsPrincipal nesnesini döndürür.
    /// </summary>
    private ClaimsPrincipal? User =>
        _httpContextAccessor.HttpContext?.User;

    /// <inheritdoc />
    public Guid? UserId
    {
        get
        {
            var value = GetClaimValue(ClaimTypes.NameIdentifier);

            return Guid.TryParse(value, out var userId)
                ? userId
                : null;
        }
    }

    /// <inheritdoc />
    public string? UserName =>
        GetClaimValue(ClaimTypes.Name);

    /// <inheritdoc />
    public string? Email =>
        GetClaimValue(ClaimTypes.Email);

    /// <inheritdoc />
    public string? FirstName =>
        GetClaimValue(ClaimTypes.GivenName);

    /// <inheritdoc />
    public string? LastName =>
        GetClaimValue(ClaimTypes.Surname);

    /// <inheritdoc />
    public string? FullName
    {
        get
        {
            var parts = new[]
                {
                    FirstName,
                    LastName
                }
                .Where(value => !string.IsNullOrWhiteSpace(value));

            var fullName = string.Join(" ", parts);

            return string.IsNullOrWhiteSpace(fullName)
                ? null
                : fullName;
        }
    }

    /// <inheritdoc />
    public Guid? OrganizationUnitId
    {
        get
        {
            var value = GetClaimValue(
                CustomClaimTypes.OrganizationUnitId);

            return Guid.TryParse(value, out var organizationUnitId)
                ? organizationUnitId
                : null;
        }
    }

    /// <inheritdoc />
    public IReadOnlyCollection<string> Roles =>
        GetClaimValues(ClaimTypes.Role);

    /// <inheritdoc />
    public IReadOnlyCollection<string> Permissions =>
        GetClaimValues(CustomClaimTypes.Permission);

    /// <inheritdoc />
    public bool IsAuthenticated =>
        User?.Identity?.IsAuthenticated == true;

    /// <inheritdoc />
    public string? IpAddress =>
        _httpContextAccessor
            .HttpContext?
            .Connection
            .RemoteIpAddress?
            .ToString();

    /// <inheritdoc />
    public string? UserAgent =>
        _httpContextAccessor
            .HttpContext?
            .Request
            .Headers
            .UserAgent
            .FirstOrDefault();

    /// <summary>
    /// Belirtilen claim tipine ait ilk claim değerini döndürür.
    /// Claim bulunamazsa null döner.
    /// </summary>
    /// <param name="claimType">
    /// Okunacak claim'in teknik tipidir.
    /// </param>
    /// <returns>
    /// Claim değeri veya claim bulunamadığında null.
    /// </returns>
    private string? GetClaimValue(string claimType)
    {
        return User?
            .FindFirst(claimType)?
            .Value;
    }

    /// <summary>
    /// Belirtilen claim tipine ait tüm benzersiz değerleri döndürür.
    /// Rol ve permission gibi birden fazla kez bulunabilecek
    /// claim'lerin okunmasında kullanılır.
    /// </summary>
    /// <param name="claimType">
    /// Okunacak claim'in teknik tipidir.
    /// </param>
    /// <returns>
    /// Claim değerlerinin salt okunur koleksiyonunu döndürür.
    /// </returns>
    private IReadOnlyCollection<string> GetClaimValues(
        string claimType)
    {
        return User?
                   .FindAll(claimType)
                   .Select(claim => claim.Value)
                   .Where(value => !string.IsNullOrWhiteSpace(value))
                   .Distinct(StringComparer.OrdinalIgnoreCase)
                   .ToArray()
               ?? Array.Empty<string>();
    }
}

Fakat permission konusunda bir önemli nokta
Permissions bilgisini ICurrentUserService'e koymamız iyi; ama ileride yazacağımız PermissionChecker bunu körü körüne tek gerçek kaynak olarak kabul etmemeli.
Çünkü örneğin kullanıcı sisteme giriş yaptı:
JWT
Users.Delete = ALLOW

sonra admin bu kullanıcıya:
UserPermission
Users.Delete = false

verdi.
Kullanıcının elindeki access token hâlâ eski permission'ı taşıyor olabilir.
Bu nedenle final authorization mimarisinde:

    → request/user context

IPermissionChecker
    → gerçek authorization kararı

ayrımını koruyacağız.
Özellikle hassas işlemlerde PermissionChecker bizim daha önce belirlediğimiz sırayı uygulayacak:
1. ADMIN?
      → ALLOW

2. UserPermission var mı?
      → IsGranted

3. RolePermission var mı?
      → ALLOW

4. Yoksa
      → DENY

Böylece CurrentUserService zengin kullanıcı bağlamı, PermissionChecker ise yetkinin gerçek karar mekanizması olacak. Bence bu ayrımla hem istediğimiz kadar bilgiye kolay erişiriz hem de güvenlik tarafını gevşetmemiş oluruz.

using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Persistence.Contexts;
using MSKBS.Persistence.Interceptors;

namespace MSKBS.Persistence;

/// <summary>
/// Persistence katmanında kullanılan veritabanı servislerini
/// dependency injection container'a kaydeder.
/// </summary>
public static class DependencyInjection
{
    /// <summary>
    /// PostgreSQL DbContext, persistence abstraction'ları ve
    /// EF Core interceptor servislerini uygulamaya ekler.
    /// </summary>
    /// <param name="services">
    /// Uygulamanın dependency injection servis koleksiyonudur.
    /// </param>
    /// <param name="configuration">
    /// Uygulama configuration bilgilerini içerir.
    /// </param>
    /// <returns>
    /// Zincirleme servis kayıtları için IServiceCollection döndürür.
    /// </returns>
    public static IServiceCollection AddPersistence(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        var connectionString =
            configuration.GetConnectionString("PostgreSQL")
            ?? throw new InvalidOperationException(
                "PostgreSQL connection string bulunamadı.");

        /*
         * Audit, soft delete ve kullanıcı bazlı CreatedBy / UpdatedBy /
         * DeletedBy işlemlerini gerçekleştiren interceptor.
         */
        services.AddScoped<AuditableEntitySaveChangesInterceptor>();

        /*
         * Uygulamanın ana PostgreSQL DbContext kaydı.
         */
        services.AddDbContext<MSKBSDbContext>(
            (serviceProvider, options) =>
            {
                options.UseNpgsql(
                    connectionString,
                    npgsqlOptions =>
                    {
                        npgsqlOptions.MigrationsAssembly(
                            typeof(MSKBSDbContext)
                                .Assembly
                                .FullName);

                        /*
                         * Geçici network/database hatalarında EF Core'un
                         * belirli sayıda tekrar denemesini sağlar.
                         */
                        npgsqlOptions.EnableRetryOnFailure(
                            maxRetryCount: 5,
                            maxRetryDelay: TimeSpan.FromSeconds(10),
                            errorCodesToAdd: null);
                    });

                options.AddInterceptors(
                    serviceProvider.GetRequiredService<
                        AuditableEntitySaveChangesInterceptor>());
            });

        /*
         * Application katmanındaki IMSKBSDbContext bağımlılığının
         * aynı scoped MSKBSDbContext instance'ını kullanmasını sağlar.
         */
        services.AddScoped<IMSKBSDbContext>(
            serviceProvider =>
                serviceProvider.GetRequiredService<MSKBSDbContext>());

        return services;
    }
}

using Microsoft.Extensions.DependencyInjection;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Infrastructure.Identity;

namespace MSKBS.Infrastructure;

/// <summary>
/// Infrastructure katmanında kullanılan teknik servisleri
/// dependency injection container'a kaydeder.
/// </summary>
public static class DependencyInjection
{
    /// <summary>
    /// HTTP context, current user ve ileride eklenecek dış servis
    /// altyapılarını uygulamaya kaydeder.
    /// </summary>
    /// <param name="services">
    /// Uygulamanın dependency injection servis koleksiyonudur.
    /// </param>
    /// <returns>
    /// Zincirleme servis kayıtları için IServiceCollection döndürür.
    /// </returns>
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services)
    {
        /*
         * HTTP request içerisindeki kullanıcı ve istemci bilgilerine
         * erişebilmek için IHttpContextAccessor kaydedilir.
         */
        services.AddHttpContextAccessor();

        /*
         * JWT/Claims içerisindeki mevcut kullanıcı bilgilerini
         * uygulamaya sağlayan servis.
         */
        services.AddScoped<ICurrentUserService, CurrentUserService>();

        return services;
    }
}

MSKBS.Persistence/Seeds/PermissionSeeder.cs
using Microsoft.EntityFrameworkCore;
using MSKBS.Domain.Entities.Identity;
using MSKBS.Persistence.Contexts;

namespace MSKBS.Persistence.Seeds;

/// <summary>
/// Uygulamada tanımlı permission kayıtlarını veritabanı ile
/// senkronize eden seed servisidir.
/// </summary>
internal sealed class PermissionSeeder
{
    private readonly MSKBSDbContext _dbContext;

    /// <summary>
    /// PermissionSeeder sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="dbContext">
    /// Permission kayıtlarının saklanacağı uygulama DbContext'idir.
    /// </param>
    public PermissionSeeder(MSKBSDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    /// <summary>
    /// Kod tarafında tanımlı permission kayıtlarını veritabanına ekler
    /// veya mevcut kayıtların metadata bilgilerini günceller.
    /// </summary>
    /// <param name="cancellationToken">
    /// İşlemin iptal edilmesini sağlayan cancellation token'dır.
    /// </param>
    public async Task SeedAsync(
        CancellationToken cancellationToken = default)
    {
        /*
         * Seed işlemi soft delete edilmiş kayıtları da görebilmelidir.
         *
         * Örneğin permission daha önce yanlışlıkla silinmişse,
         * yeni bir duplicate kayıt oluşturmak yerine mevcut kaydı
         * tekrar aktif hale getiriyoruz.
         */
        var existingPermissions = await _dbContext.Permissions
            .IgnoreQueryFilters()
            .ToDictionaryAsync(
                permission => permission.Code,
                StringComparer.OrdinalIgnoreCase,
                cancellationToken);

        foreach (var seedItem in IdentitySeedData.Permissions)
        {
            if (existingPermissions.TryGetValue(
                    seedItem.Code,
                    out var permission))
            {
                UpdatePermission(permission, seedItem);
                continue;
            }

            permission = new Permission
            {
                Name = seedItem.Name,
                Code = seedItem.Code,
                Module = seedItem.Module,
                SortOrder = seedItem.SortOrder,
                Description = null
            };

            await _dbContext.Permissions.AddAsync(
                permission,
                cancellationToken);
        }

        await _dbContext.SaveChangesAsync(cancellationToken);
    }

    /// <summary>
    /// Mevcut permission kaydını kod tarafındaki güncel seed
    /// tanımıyla senkronize eder.
    /// </summary>
    private static void UpdatePermission(
        Permission permission,
        PermissionSeedItem seedItem)
    {
        permission.Name = seedItem.Name;
        permission.Module = seedItem.Module;
        permission.SortOrder = seedItem.SortOrder;

        /*
         * Sistemin zorunlu permission'ı daha önce soft delete veya
         * pasif yapılmışsa seed sırasında tekrar kullanıma alınır.
         */
        permission.IsDeleted = false;
        permission.IsActive = true;
        permission.DeletedDate = null;
        permission.DeletedBy = null;
    }
}

4. PasswordHasherService
ASP.NET Core'un standart password hasher'ını kullanalım.
MSKBS.Infrastructure/Identity/PasswordHasherService.cs
using Microsoft.AspNetCore.Identity;
using MSKBS.Application.Common.Interfaces.Identity;

namespace MSKBS.Infrastructure.Identity;

/// <summary>
/// ASP.NET Core Identity password hashing altyapısını kullanarak
/// parola hashleme ve doğrulama işlemlerini gerçekleştirir.
/// </summary>
public sealed class PasswordHasherService : IPasswordHasherService
{
    private readonly PasswordHasher<object> _passwordHasher = new();

    /// <inheritdoc />
    public string HashPassword(string password)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(password);

        return _passwordHasher.HashPassword(
            user: null!,
            password);
    }

    /// <inheritdoc />
    public bool VerifyPassword(
        string hashedPassword,
        string providedPassword)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(hashedPassword);
        ArgumentException.ThrowIfNullOrWhiteSpace(providedPassword);

        var result = _passwordHasher.VerifyHashedPassword(
            user: null!,
            hashedPassword,
            providedPassword);

        return result is
            PasswordVerificationResult.Success
            or PasswordVerificationResult.SuccessRehashNeeded;
    }
}

Böylece kendi SHA256 parola sistemi gibi yanlış bir yöntem yapmıyoruz.

Parola:
PasswordHasher / PBKDF tabanlı parola hashing

Refresh token:
SHA-256 gibi deterministic hash

olacak.

6. IdentitySeeder.cs
Şimdi roller + role permissions + ilk admin.
MSKBS.Persistence/Seeds/IdentitySeeder.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Domain.Constants;
using MSKBS.Domain.Entities.Identity;
using MSKBS.Persistence.Contexts;

namespace MSKBS.Persistence.Seeds;

/// <summary>
/// MSKBS Identity altyapısının başlangıç rollerini,
/// rol-permission ilişkilerini ve ilk yönetici kullanıcısını oluşturur.
/// </summary>
public sealed class IdentitySeeder
{
    private readonly MSKBSDbContext _dbContext;
    private readonly PermissionSeeder _permissionSeeder;
    private readonly IPasswordHasherService _passwordHasherService;
    private readonly IConfiguration _configuration;

    /// <summary>
    /// IdentitySeeder sınıfının yeni örneğini oluşturur.
    /// </summary>
    public IdentitySeeder(
        MSKBSDbContext dbContext,
        PermissionSeeder permissionSeeder,
        IPasswordHasherService passwordHasherService,
        IConfiguration configuration)
    {
        _dbContext = dbContext;
        _permissionSeeder = permissionSeeder;
        _passwordHasherService = passwordHasherService;
        _configuration = configuration;
    }

    /// <summary>
    /// Identity sisteminin başlangıç verilerini idempotent şekilde oluşturur.
    /// </summary>
    /// <param name="cancellationToken">
    /// İşlemin iptal edilmesini sağlayan cancellation token'dır.
    /// </param>
    public async Task SeedAsync(
        CancellationToken cancellationToken = default)
    {
        await _permissionSeeder.SeedAsync(cancellationToken);

        await SeedRolesAsync(cancellationToken);

        await AssignAdminPermissionsAsync(cancellationToken);

        await SeedInitialAdminAsync(cancellationToken);
    }

    /// <summary>
    /// Başlangıç rollerini oluşturur veya mevcut sistem rollerini
    /// güncel seed tanımlarıyla senkronize eder.
    /// </summary>
    private async Task SeedRolesAsync(
        CancellationToken cancellationToken)
    {
        var existingRoles = await _dbContext.Roles
            .IgnoreQueryFilters()
            .ToDictionaryAsync(
                role => role.Code,
                StringComparer.OrdinalIgnoreCase,
                cancellationToken);

        foreach (var seedItem in IdentitySeedData.Roles)
        {
            if (existingRoles.TryGetValue(
                    seedItem.Code,
                    out var role))
            {
                role.Name = seedItem.Name;
                role.Description = seedItem.Description;
                role.IsSystemRole = seedItem.IsSystemRole;

                role.IsActive = true;
                role.IsDeleted = false;
                role.DeletedDate = null;
                role.DeletedBy = null;

                continue;
            }

            role = new Role
            {
                Name = seedItem.Name,
                Code = seedItem.Code,
                Description = seedItem.Description,
                IsSystemRole = seedItem.IsSystemRole
            };

            await _dbContext.Roles.AddAsync(
                role,
                cancellationToken);
        }

        await _dbContext.SaveChangesAsync(cancellationToken);
    }

    /// <summary>
    /// ADMIN rolüne sistemde bulunan bütün permission kayıtlarını atar.
    ///
    /// ADMIN authorization sırasında zaten özel olarak tam yetkili kabul
    /// edilecek olsa da rol-permission ilişkilerinin veritabanında tutulması
    /// yönetim ekranları ve görünürlük açısından faydalıdır.
    /// </summary>
    private async Task AssignAdminPermissionsAsync(
        CancellationToken cancellationToken)
    {
        var adminRole = await _dbContext.Roles
            .SingleAsync(
                role => role.Code == SystemRoles.Admin,
                cancellationToken);

        var permissions = await _dbContext.Permissions
            .ToListAsync(cancellationToken);

        var existingPermissionIds = await _dbContext.RolePermissions
            .Where(rolePermission =>
                rolePermission.RoleId == adminRole.Id)
            .Select(rolePermission =>
                rolePermission.PermissionId)
            .ToHashSetAsync(cancellationToken);

        foreach (var permission in permissions)
        {
            if (existingPermissionIds.Contains(permission.Id))
            {
                continue;
            }

            await _dbContext.RolePermissions.AddAsync(
                new RolePermission
                {
                    RoleId = adminRole.Id,
                    PermissionId = permission.Id
                },
                cancellationToken);
        }

        await _dbContext.SaveChangesAsync(cancellationToken);
    }

    /// <summary>
    /// Configuration üzerinden sağlanan bilgilerle ilk yönetici
    /// kullanıcısını oluşturur ve ADMIN rolünü atar.
    /// </summary>
    private async Task SeedInitialAdminAsync(
        CancellationToken cancellationToken)
    {
        var userName =
            _configuration["InitialAdmin:UserName"];

        var email =
            _configuration["InitialAdmin:Email"];

        var password =
            _configuration["InitialAdmin:Password"];

        /*
         * Initial admin bilgileri yapılandırılmadıysa kullanıcı seed işlemi
         * yapılmaz. Böylece source code içerisinde varsayılan production
         * parolası bulundurma zorunluluğu oluşmaz.
         */
        if (string.IsNullOrWhiteSpace(userName) ||
            string.IsNullOrWhiteSpace(email) ||
            string.IsNullOrWhiteSpace(password))
        {
            return;
        }

        var normalizedUserName =
            Normalize(userName);

        var normalizedEmail =
            Normalize(email);

        var existingUser = await _dbContext.Users
            .IgnoreQueryFilters()
            .FirstOrDefaultAsync(
                user =>
                    user.NormalizedUserName == normalizedUserName ||
                    user.NormalizedEmail == normalizedEmail,
                cancellationToken);

        User adminUser;

        if (existingUser is null)
        {
            adminUser = new User
            {
                UserName = userName.Trim(),
                NormalizedUserName = normalizedUserName,

                Email = email.Trim(),
                NormalizedEmail = normalizedEmail,

                FirstName =
                    _configuration["InitialAdmin:FirstName"]?.Trim()
                    ?? "System",

                LastName =
                    _configuration["InitialAdmin:LastName"]?.Trim()
                    ?? "Administrator",

                PasswordHash =
                    _passwordHasherService.HashPassword(password),

                SecurityStamp =
                    Guid.NewGuid().ToString("N"),

                PasswordChangedDate =
                    DateTime.UtcNow
            };

            await _dbContext.Users.AddAsync(
                adminUser,
                cancellationToken);

            await _dbContext.SaveChangesAsync(cancellationToken);
        }
        else
        {
            adminUser = existingUser;

            /*
             * Seeder her startup'ta admin parolasını değiştirmemelidir.
             * Mevcut kullanıcı bulunduğunda yalnızca hesabı tekrar
             * kullanılabilir durumda tutuyoruz.
             */
            adminUser.IsDeleted = false;
            adminUser.IsActive = true;
            adminUser.DeletedDate = null;
            adminUser.DeletedBy = null;
        }

        await AssignAdminRoleAsync(
            adminUser,
            cancellationToken);
    }

    /// <summary>
    /// Belirtilen kullanıcıya ADMIN rolünü atar.
    /// Mevcut ilişki varsa duplicate kayıt oluşturmaz.
    /// </summary>
    private async Task AssignAdminRoleAsync(
        User user,
        CancellationToken cancellationToken)
    {
        var adminRole = await _dbContext.Roles
            .SingleAsync(
                role => role.Code == SystemRoles.Admin,
                cancellationToken);

        var existingUserRole = await _dbContext.UserRoles
            .IgnoreQueryFilters()
            .FirstOrDefaultAsync(
                userRole =>
                    userRole.UserId == user.Id &&
                    userRole.RoleId == adminRole.Id,
                cancellationToken);

        if (existingUserRole is not null)
        {
            existingUserRole.IsActive = true;
            existingUserRole.IsDeleted = false;
            existingUserRole.DeletedDate = null;
            existingUserRole.DeletedBy = null;

            await _dbContext.SaveChangesAsync(cancellationToken);

            return;
        }

        await _dbContext.UserRoles.AddAsync(
            new UserRole
            {
                UserId = user.Id,
                RoleId = adminRole.Id
            },
            cancellationToken);

        await _dbContext.SaveChangesAsync(cancellationToken);
    }

    /// <summary>
    /// Kullanıcı adı ve e-posta gibi karşılaştırılabilir kimlik
    /// değerlerini kültürden bağımsız şekilde normalize eder.
    /// </summary>
    private static string Normalize(string value)
    {
        return value
            .Trim()
            .ToUpperInvariant();
    }
}

using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using MSKBS.Persistence.Contexts;
using MSKBS.Persistence.Seeds;

namespace MSKBS.Persistence.Extensions;

/// <summary>
/// MSKBS veritabanının migration ve başlangıç seed işlemlerini
/// gerçekleştiren extension metotlarını içerir.
/// </summary>
public static class DatabaseInitializationExtensions
{
    /// <summary>
    /// Bekleyen EF Core migration'larını uygular ve Identity
    /// başlangıç verilerini oluşturur.
    /// </summary>
    /// <param name="serviceProvider">
    /// Uygulamanın root service provider nesnesidir.
    /// </param>
    /// <param name="cancellationToken">
    /// İşlemin iptal edilmesini sağlayan cancellation token'dır.
    /// </param>
    public static async Task InitializeDatabaseAsync(
        this IServiceProvider serviceProvider,
        CancellationToken cancellationToken = default)
    {
        await using var scope =
            serviceProvider.CreateAsyncScope();

        var dbContext =
            scope.ServiceProvider
                .GetRequiredService<MSKBSDbContext>();

        await dbContext.Database.MigrateAsync(
            cancellationToken);

        var identitySeeder =
            scope.ServiceProvider
                .GetRequiredService<IdentitySeeder>();

        await identitySeeder.SeedAsync(
            cancellationToken);
    }
}

7. JwtTokenService.cs
Ana servisimiz:
MSKBS.Infrastructure/Authentication/JwtTokenService.cs
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Security.Cryptography;
using System.Text;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.Tokens;
using MSKBS.Application.Common.Constants;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Models.Authentication;
using MSKBS.Domain.Entities.Identity;

namespace MSKBS.Infrastructure.Authentication;

/// <summary>
/// HMAC SHA-256 imzalama algoritması kullanarak JWT access token
/// ve kriptografik olarak güvenli refresh token oluşturan servistir.
/// </summary>
public sealed class JwtTokenService : IJwtTokenService
{
    private const int RefreshTokenByteLength = 64;

    private readonly JwtSettings _settings;
    private readonly TimeProvider _timeProvider;
    private readonly SigningCredentials _signingCredentials;

    /// <summary>
    /// JwtTokenService sınıfının yeni bir örneğini oluşturur.
    /// </summary>
    /// <param name="options">
    /// JWT configuration ayarlarını sağlar.
    /// </param>
    /// <param name="timeProvider">
    /// UTC zaman bilgisinin test edilebilir şekilde alınmasını sağlar.
    /// </param>
    public JwtTokenService(
        IOptions<JwtSettings> options,
        TimeProvider timeProvider)
    {
        _settings = options.Value;
        _timeProvider = timeProvider;

        var securityKey = new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes(_settings.SecretKey));

        _signingCredentials = new SigningCredentials(
            securityKey,
            SecurityAlgorithms.HmacSha256);
    }

    /// <inheritdoc />
    public AccessTokenResult GenerateAccessToken(
        User user,
        IReadOnlyCollection<string> roles,
        IReadOnlyCollection<string> permissions)
    {
        ArgumentNullException.ThrowIfNull(user);
        ArgumentNullException.ThrowIfNull(roles);
        ArgumentNullException.ThrowIfNull(permissions);

        var utcNow = _timeProvider.GetUtcNow().UtcDateTime;

        var expiresAt = utcNow.AddMinutes(
            _settings.AccessTokenExpirationMinutes);

        var claims = CreateUserClaims(
            user,
            roles,
            permissions);

        var token = new JwtSecurityToken(
            issuer: _settings.Issuer,
            audience: _settings.Audience,
            claims: claims,
            notBefore: utcNow,
            expires: expiresAt,
            signingCredentials: _signingCredentials);

        var tokenHandler = new JwtSecurityTokenHandler();

        return new AccessTokenResult
        {
            Token = tokenHandler.WriteToken(token),
            ExpiresAt = expiresAt
        };
    }

    /// <inheritdoc />
    public RefreshTokenResult GenerateRefreshToken()
    {
        var randomBytes =
            RandomNumberGenerator.GetBytes(RefreshTokenByteLength);

        /*
         * Base64Url formatı URL, JSON ve HTTP header/cookie gibi
         * taşıma ortamlarında standart Base64'e göre daha güvenlidir.
         */
        var token = Base64UrlEncoder.Encode(randomBytes);

        var expiresAt = _timeProvider
            .GetUtcNow()
            .UtcDateTime
            .AddDays(_settings.RefreshTokenExpirationDays);

        return new RefreshTokenResult
        {
            Token = token,
            TokenHash = HashRefreshToken(token),
            ExpiresAt = expiresAt
        };
    }

    /// <inheritdoc />
    public string HashRefreshToken(string token)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(token);

        var tokenBytes = Encoding.UTF8.GetBytes(token);
        var hashBytes = SHA256.HashData(tokenBytes);

        return Convert.ToHexString(hashBytes);
    }

    /// <summary>
    /// Kullanıcının kimlik, organizasyon, rol ve permission
    /// bilgilerini JWT claim koleksiyonuna dönüştürür.
    /// </summary>
    /// <param name="user">
    /// Token oluşturulan kullanıcıdır.
    /// </param>
    /// <param name="roles">
    /// Kullanıcının rol kodlarıdır.
    /// </param>
    /// <param name="permissions">
    /// Kullanıcının effective permission kodlarıdır.
    /// </param>
    /// <returns>
    /// JWT içerisine yazılacak claim koleksiyonunu döndürür.
    /// </returns>
    private static List<Claim> CreateUserClaims(
        User user,
        IReadOnlyCollection<string> roles,
        IReadOnlyCollection<string> permissions)
    {
        var claims = new List<Claim>
        {
            new(
                JwtRegisteredClaimNames.Jti,
                Guid.NewGuid().ToString("N")),

            new(
                ClaimTypes.NameIdentifier,
                user.Id.ToString()),

            new(
                ClaimTypes.Name,
                user.UserName),

            new(
                ClaimTypes.Email,
                user.Email),

            new(
                ClaimTypes.GivenName,
                user.FirstName),

            new(
                ClaimTypes.Surname,
                user.LastName)
        };

        if (user.OrganizationUnitId.HasValue)
        {
            claims.Add(
                new Claim(
                    CustomClaimTypes.OrganizationUnitId,
                    user.OrganizationUnitId.Value.ToString()));
        }

        foreach (var role in roles
                     .Where(role => !string.IsNullOrWhiteSpace(role))
                     .Distinct(StringComparer.OrdinalIgnoreCase))
        {
            claims.Add(
                new Claim(
                    ClaimTypes.Role,
                    role));
        }

        foreach (var permission in permissions
                     .Where(permission =>
                         !string.IsNullOrWhiteSpace(permission))
                     .Distinct(StringComparer.OrdinalIgnoreCase))
        {
            claims.Add(
                new Claim(
                    CustomClaimTypes.Permission,
                    permission));
        }

        return claims;
    }
}

using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Infrastructure.Authentication;
using MSKBS.Infrastructure.Identity;

namespace MSKBS.Infrastructure;

/// <summary>
/// Infrastructure katmanındaki authentication, identity ve diğer teknik
/// servisleri dependency injection container'a kaydeder.
/// </summary>
public static class DependencyInjection
{
    /// <summary>
    /// Infrastructure katmanında kullanılan servisleri ve JWT
    /// configuration altyapısını uygulamaya ekler.
    /// </summary>
    /// <param name="services">
    /// Uygulamanın dependency injection servis koleksiyonudur.
    /// </param>
    /// <param name="configuration">
    /// Uygulamanın configuration bilgilerini sağlar.
    /// </param>
    /// <returns>
    /// Zincirleme servis kayıtları için IServiceCollection döndürür.
    /// </returns>
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        services.AddHttpContextAccessor();

        services.AddSingleton(TimeProvider.System);

        services.AddScoped<
            ICurrentUserService,
            CurrentUserService>();

        services.AddScoped<
            IPasswordHasherService,
            PasswordHasherService>();

        services.AddScoped<
            IJwtTokenService,
            JwtTokenService>();

        services
            .AddOptions<JwtSettings>()
            .Bind(
                configuration.GetSection(
                    JwtSettings.SectionName))
            .Validate(
                settings =>
                    !string.IsNullOrWhiteSpace(settings.SecretKey),
                "Jwt:SecretKey boş bırakılamaz.")
            .Validate(
                settings =>
                    Encoding.UTF8.GetByteCount(settings.SecretKey) >= 32,
                "Jwt:SecretKey en az 32 byte uzunluğunda olmalıdır.")
            .Validate(
                settings =>
                    !string.IsNullOrWhiteSpace(settings.Issuer),
                "Jwt:Issuer boş bırakılamaz.")
            .Validate(
                settings =>
                    !string.IsNullOrWhiteSpace(settings.Audience),
                "Jwt:Audience boş bırakılamaz.")
            .Validate(
                settings =>
                    settings.AccessTokenExpirationMinutes > 0,
                "Access token süresi sıfırdan büyük olmalıdır.")
            .Validate(
                settings =>
                    settings.RefreshTokenExpirationDays > 0,
                "Refresh token süresi sıfırdan büyük olmalıdır.")
            .ValidateOnStart();

        return services;
    }
}

3. UserAuthorizationService
Bunu Application katmanında tutabiliriz çünkü doğrudan IMSKBSDbContext abstraction’ı üzerinden çalışıyor.
MSKBS.Application/Common/Authorization/UserAuthorizationService.cs
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Interfaces.Authorization;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Domain.Constants;

namespace MSKBS.Application.Common.Authorization;

/// <summary>
/// Kullanıcının rollerini ve kullanıcı bazlı permission override
/// kayıtlarını değerlendirerek effective permission setini hesaplar.
/// </summary>
public sealed class UserAuthorizationService
    : IUserAuthorizationService
{
    private readonly IMSKBSDbContext _dbContext;

    /// <summary>
    /// UserAuthorizationService sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="dbContext">
    /// Identity ve permission kayıtlarına erişim sağlayan DbContext'tir.
    /// </param>
    public UserAuthorizationService(
        IMSKBSDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    /// <inheritdoc />
    public async Task<UserAuthorizationSnapshot> GetAsync(
        Guid userId,
        CancellationToken cancellationToken = default)
    {
        var roles = await _dbContext.UserRoles
            .Where(userRole => userRole.UserId == userId)
            .Select(userRole => userRole.Role.Code)
            .Distinct()
            .ToListAsync(cancellationToken);

        var isAdmin = roles.Contains(
            SystemRoles.Admin,
            StringComparer.OrdinalIgnoreCase);

        /*
         * ADMIN tüm permission'lara sahip kabul edilir.
         *
         * Token içerisinde de bütün permission kodlarını gönderiyoruz.
         * Böylece Angular tarafındaki permission görünürlük kontrolleri
         * ayrıca özel ADMIN kontrolüne ihtiyaç duymaz.
         */
        if (isAdmin)
        {
            var allPermissions = await _dbContext.Permissions
                .Select(permission => permission.Code)
                .Distinct()
                .ToListAsync(cancellationToken);

            return new UserAuthorizationSnapshot
            {
                Roles = roles,
                Permissions = allPermissions,
                IsAdmin = true
            };
        }

        var rolePermissions = await _dbContext.RolePermissions
            .Where(rolePermission =>
                _dbContext.UserRoles.Any(userRole =>
                    userRole.UserId == userId &&
                    userRole.RoleId == rolePermission.RoleId))
            .Select(rolePermission =>
                rolePermission.Permission.Code)
            .Distinct()
            .ToListAsync(cancellationToken);

        var effectivePermissions =
            new HashSet<string>(
                rolePermissions,
                StringComparer.OrdinalIgnoreCase);

        var userOverrides = await _dbContext.UserPermissions
            .Where(userPermission =>
                userPermission.UserId == userId)
            .Select(userPermission => new
            {
                userPermission.Permission.Code,
                userPermission.IsGranted
            })
            .ToListAsync(cancellationToken);

        foreach (var userOverride in userOverrides)
        {
            if (userOverride.IsGranted)
            {
                effectivePermissions.Add(
                    userOverride.Code);
            }
            else
            {
                effectivePermissions.Remove(
                    userOverride.Code);
            }
        }

        return new UserAuthorizationSnapshot
        {
            Roles = roles,
            Permissions = effectivePermissions.ToArray(),
            IsAdmin = false
        };
    }
}

10. LoginCommandHandler
Şimdi ana parça.
using MediatR;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;
using MSKBS.Application.Authentication.Responses;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Interfaces.Authorization;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Application.Common.Options;
using MSKBS.Domain.Entities.Identity;

namespace MSKBS.Application.Authentication.Commands.Login;

/// <summary>
/// Kullanıcı giriş bilgilerini doğrular, güvenlik kontrollerini uygular,
/// authorization bilgilerini hesaplar ve access/refresh token üretir.
/// </summary>
public sealed class LoginCommandHandler
    : IRequestHandler<LoginCommand, LoginResult>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly IPasswordHasherService _passwordHasherService;
    private readonly IJwtTokenService _jwtTokenService;
    private readonly IUserAuthorizationService _authorizationService;
    private readonly ICurrentUserService _currentUserService;
    private readonly AuthenticationSettings _authenticationSettings;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// LoginCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public LoginCommandHandler(
        IMSKBSDbContext dbContext,
        IPasswordHasherService passwordHasherService,
        IJwtTokenService jwtTokenService,
        IUserAuthorizationService authorizationService,
        ICurrentUserService currentUserService,
        IOptions<AuthenticationSettings> authenticationSettings,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _passwordHasherService = passwordHasherService;
        _jwtTokenService = jwtTokenService;
        _authorizationService = authorizationService;
        _currentUserService = currentUserService;
        _authenticationSettings = authenticationSettings.Value;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task<LoginResult> Handle(
        LoginCommand request,
        CancellationToken cancellationToken)
    {
        var utcNow =
            _timeProvider.GetUtcNow().UtcDateTime;

        var normalizedIdentifier =
            Normalize(request.Identifier);

        /*
         * Normal global filters sayesinde soft delete edilmiş veya
         * pasif kullanıcılar authentication sorgusuna dahil edilmez.
         */
        var user = await _dbContext.Users
            .FirstOrDefaultAsync(
                user =>
                    user.NormalizedUserName == normalizedIdentifier ||
                    user.NormalizedEmail == normalizedIdentifier,
                cancellationToken);

        if (user is null)
        {
            throw new LoginFailedException();
        }

        ValidateAccountPeriod(
            user,
            utcNow);

        HandleExpiredLockout(
            user,
            utcNow);

        if (user.IsLocked)
        {
            throw new AccountLockedException(
                user.LockoutEnd);
        }

        var passwordValid =
            _passwordHasherService.VerifyPassword(
                user.PasswordHash,
                request.Password);

        if (!passwordValid)
        {
            await HandleFailedLoginAsync(
                user,
                utcNow,
                cancellationToken);

            throw new LoginFailedException();
        }

        ResetFailedLoginState(user);

        user.LastLoginDate = utcNow;

        var authorization =
            await _authorizationService.GetAsync(
                user.Id,
                cancellationToken);

        var accessToken =
            _jwtTokenService.GenerateAccessToken(
                user,
                authorization.Roles,
                authorization.Permissions);

        var refreshToken =
            _jwtTokenService.GenerateRefreshToken();

        var refreshTokenEntity =
            new RefreshToken
            {
                UserId = user.Id,
                TokenHash = refreshToken.TokenHash,
                ExpiresAt = refreshToken.ExpiresAt,

                CreatedByIp =
                    _currentUserService.IpAddress,

                UserAgent =
                    _currentUserService.UserAgent,

                DeviceName =
                    string.IsNullOrWhiteSpace(request.DeviceName)
                        ? "Web"
                        : request.DeviceName.Trim()
            };

        await _dbContext.RefreshTokens.AddAsync(
            refreshTokenEntity,
            cancellationToken);

        await _dbContext.SaveChangesAsync(
            cancellationToken);

        return new LoginResult
        {
            RefreshToken = refreshToken.Token,
            RefreshTokenExpiresAt = refreshToken.ExpiresAt,

            Response = new LoginResponse
            {
                AccessToken = accessToken.Token,
                AccessTokenExpiresAt =
                    accessToken.ExpiresAt,

                User = new LoginUserResponse
                {
                    Id = user.Id,
                    UserName = user.UserName,
                    Email = user.Email,
                    FirstName = user.FirstName,
                    LastName = user.LastName,

                    OrganizationUnitId =
                        user.OrganizationUnitId,

                    Roles = authorization.Roles,
                    Permissions =
                        authorization.Permissions
                }
            }
        };
    }

    /// <summary>
    /// Kullanıcı hesabının geçerlilik başlangıç ve bitiş
    /// tarihlerini kontrol eder.
    /// </summary>
    private static void ValidateAccountPeriod(
        User user,
        DateTime utcNow)
    {
        if (user.ValidFrom.HasValue &&
            user.ValidFrom.Value > utcNow)
        {
            throw new LoginFailedException();
        }

        if (user.ValidTo.HasValue &&
            user.ValidTo.Value <= utcNow)
        {
            throw new LoginFailedException();
        }
    }

    /// <summary>
    /// Süresi dolmuş geçici hesap kilidini kaldırır.
    /// Süresiz manuel kilitlere müdahale etmez.
    /// </summary>
    private static void HandleExpiredLockout(
        User user,
        DateTime utcNow)
    {
        if (!user.IsLocked ||
            !user.LockoutEnd.HasValue ||
            user.LockoutEnd.Value > utcNow)
        {
            return;
        }

        user.IsLocked = false;
        user.LockoutEnd = null;
        user.AccessFailedCount = 0;
    }

    /// <summary>
    /// Başarısız parola doğrulamasını kaydeder ve belirlenen
    /// limite ulaşıldığında hesabı geçici olarak kilitler.
    /// </summary>
    private async Task HandleFailedLoginAsync(
        User user,
        DateTime utcNow,
        CancellationToken cancellationToken)
    {
        user.AccessFailedCount++;

        if (user.AccessFailedCount >=
            _authenticationSettings.MaxFailedAccessAttempts)
        {
            user.IsLocked = true;

            user.LockoutEnd =
                utcNow.AddMinutes(
                    _authenticationSettings.LockoutMinutes);
        }

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }

    /// <summary>
    /// Başarılı authentication sonrasında başarısız giriş
    /// ve geçici kilit bilgilerini sıfırlar.
    /// </summary>
    private static void ResetFailedLoginState(
        User user)
    {
        user.AccessFailedCount = 0;
        user.IsLocked = false;
        user.LockoutEnd = null;
    }

    /// <summary>
    /// Kullanıcı adı veya e-posta adresini veritabanındaki
    /// normalize edilmiş kolonlarla karşılaştırılabilir hale getirir.
    /// </summary>
    private static string Normalize(
        string value)
    {
        return value
            .Trim()
            .ToUpperInvariant();
    }
}

using MediatR;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using MSKBS.Application.Authentication.Commands.Login;
using MSKBS.Application.Authentication.Responses;

namespace MSKBS.Api.Controllers;

/// <summary>
/// Kullanıcı authentication işlemlerini sağlayan API endpoint'lerini içerir.
/// </summary>
[ApiController]
[Route("api/auth")]
public sealed class AuthController : ControllerBase
{
    private const string RefreshTokenCookieName =
        "mskbs_refresh_token";

    private readonly ISender _sender;

    /// <summary>
    /// AuthController sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="sender">
    /// CQRS command ve query işlemlerini MediatR üzerinden çalıştırır.
    /// </param>
    public AuthController(
        ISender sender)
    {
        _sender = sender;
    }

    /// <summary>
    /// Kullanıcı adı/e-posta ve parola ile sisteme giriş yapar.
    /// Başarılı girişte access token response içerisinde,
    /// refresh token ise HttpOnly cookie içerisinde gönderilir.
    /// </summary>
    /// <param name="command">
    /// Kullanıcının giriş bilgileridir.
    /// </param>
    /// <param name="cancellationToken">
    /// İstek iptal token'ıdır.
    /// </param>
    /// <returns>
    /// Access token ve kullanıcı bilgilerini döndürür.
    /// </returns>
    [HttpPost("login")]
    [AllowAnonymous]
    [ProducesResponseType(
        typeof(LoginResponse),
        StatusCodes.Status200OK)]
    public async Task<ActionResult<LoginResponse>> Login(
        [FromBody] LoginCommand command,
        CancellationToken cancellationToken)
    {
        var result = await _sender.Send(
            command,
            cancellationToken);

        AppendRefreshTokenCookie(
            result.RefreshToken,
            result.RefreshTokenExpiresAt);

        return Ok(result.Response);
    }

    /// <summary>
    /// Refresh token değerini güvenli HttpOnly cookie olarak response'a ekler.
    /// </summary>
    /// <param name="refreshToken">
    /// İstemciye gönderilecek açık refresh token değeridir.
    /// </param>
    /// <param name="expiresAt">
    /// Cookie'nin geçerliliğinin sona ereceği UTC tarih ve saattir.
    /// </param>
    private void AppendRefreshTokenCookie(
        string refreshToken,
        DateTime expiresAt)
    {
        Response.Cookies.Append(
            RefreshTokenCookieName,
            refreshToken,
            new CookieOptions
            {
                HttpOnly = true,
                Secure = true,

                /*
                 * Angular ve API aynı site altında çalışacaksa
                 * Lax iyi varsayılandır.
                 */
                SameSite = SameSiteMode.Lax,

                Expires = new DateTimeOffset(
                    DateTime.SpecifyKind(
                        expiresAt,
                        DateTimeKind.Utc)),

                IsEssential = true,

                Path = "/api/auth"
            });
    }
}

MSKBS.Application/Common/Behaviors/ValidationBehavior.cs
using FluentValidation;
using MediatR;

namespace MSKBS.Application.Common.Behaviors;

/// <summary>
/// MediatR üzerinden çalışan request modellerini ilgili FluentValidation
/// validator'ları ile doğrulayan merkezi pipeline behavior'dır.
/// </summary>
/// <typeparam name="TRequest">
/// Doğrulanacak MediatR request tipidir.
/// </typeparam>
/// <typeparam name="TResponse">
/// Request sonucunda döndürülecek response tipidir.
/// </typeparam>
public sealed class ValidationBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : notnull
{
    private readonly IEnumerable<IValidator<TRequest>> _validators;

    /// <summary>
    /// ValidationBehavior sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="validators">
    /// İlgili request tipi için tanımlanmış FluentValidation validator'larıdır.
    /// </param>
    public ValidationBehavior(
        IEnumerable<IValidator<TRequest>> validators)
    {
        _validators = validators;
    }

    /// <summary>
    /// Request handler çalıştırılmadan önce bütün validator'ları çalıştırır.
    /// Validation hatası varsa handler'a geçmeden ValidationException fırlatır.
    /// </summary>
    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        if (!_validators.Any())
        {
            return await next();
        }

        var context =
            new ValidationContext<TRequest>(request);

        var validationResults =
            await Task.WhenAll(
                _validators.Select(
                    validator =>
                        validator.ValidateAsync(
                            context,
                            cancellationToken)));

        var failures = validationResults
            .SelectMany(result => result.Errors)
            .Where(failure => failure is not null)
            .ToList();

        if (failures.Count != 0)
        {
            throw new ValidationException(failures);
        }

        return await next();
    }
}

GlobalExceptionHandler
MSKBS.Api/ExceptionHandling/GlobalExceptionHandler.cs
using System.Diagnostics;
using FluentValidation;
using Microsoft.AspNetCore.Diagnostics;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Exceptions;

namespace MSKBS.Api.ExceptionHandling;

/// <summary>
/// Uygulama genelinde oluşan exception'ları merkezi olarak yakalar
/// ve standart ProblemDetails HTTP response'larına dönüştürür.
/// </summary>
public sealed class GlobalExceptionHandler : IExceptionHandler
{
    private readonly ILogger<GlobalExceptionHandler> _logger;

    /// <summary>
    /// GlobalExceptionHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public GlobalExceptionHandler(
        ILogger<GlobalExceptionHandler> logger)
    {
        _logger = logger;
    }

    /// <summary>
    /// Oluşan exception tipine göre uygun HTTP status code ve
    /// ProblemDetails response'unu oluşturur.
    /// </summary>
    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        var problemDetails =
            CreateProblemDetails(
                httpContext,
                exception);

        if (problemDetails.Status is >= 500)
        {
            _logger.LogError(
                exception,
                "Beklenmeyen uygulama hatası oluştu. TraceId: {TraceId}",
                httpContext.TraceIdentifier);
        }
        else
        {
            _logger.LogWarning(
                "İstek başarısız oldu. Exception: {ExceptionType}, StatusCode: {StatusCode}, TraceId: {TraceId}",
                exception.GetType().Name,
                problemDetails.Status,
                httpContext.TraceIdentifier);
        }

        httpContext.Response.StatusCode =
            problemDetails.Status
            ?? StatusCodes.Status500InternalServerError;

        await httpContext.Response.WriteAsJsonAsync(
            problemDetails,
            cancellationToken);

        return true;
    }

    /// <summary>
    /// Exception tipini HTTP ProblemDetails modeline dönüştürür.
    /// </summary>
    private static ProblemDetails CreateProblemDetails(
        HttpContext httpContext,
        Exception exception)
    {
        var problemDetails = exception switch
        {
            ValidationException validationException =>
                CreateValidationProblem(
                    validationException),

            LoginFailedException =>
                new ProblemDetails
                {
                    Status = StatusCodes.Status401Unauthorized,
                    Title = "Giriş başarısız.",
                    Detail = exception.Message
                },

            InvalidRefreshTokenException =>
                new ProblemDetails
                {
                    Status = StatusCodes.Status401Unauthorized,
                    Title = "Oturum geçersiz.",
                    Detail = exception.Message
                },

            AccountLockedException =>
                new ProblemDetails
                {
                    Status = StatusCodes.Status423Locked,
                    Title = "Hesap kilitli.",
                    Detail = exception.Message
                },

            DbUpdateConcurrencyException =>
                new ProblemDetails
                {
                    Status = StatusCodes.Status409Conflict,
                    Title = "Kayıt güncellenemedi.",
                    Detail =
                        "Kayıt başka bir kullanıcı tarafından güncellenmiştir. " +
                        "Lütfen güncel veriyi tekrar yükleyiniz."
                },

            _ =>
                new ProblemDetails
                {
                    Status =
                        StatusCodes.Status500InternalServerError,

                    Title =
                        "Beklenmeyen bir hata oluştu.",

                    Detail =
                        "İşlem sırasında beklenmeyen bir hata oluştu."
                }
        };

        problemDetails.Instance =
            httpContext.Request.Path;

        problemDetails.Extensions["traceId"] =
            Activity.Current?.Id
            ?? httpContext.TraceIdentifier;

        return problemDetails;
    }

    /// <summary>
    /// FluentValidation hatalarını property bazlı validation
    /// response'una dönüştürür.
    /// </summary>
    private static ProblemDetails CreateValidationProblem(
        ValidationException exception)
    {
        var errors = exception.Errors
            .GroupBy(error => error.PropertyName)
            .ToDictionary(
                group => group.Key,
                group => group
                    .Select(error => error.ErrorMessage)
                    .Distinct()
                    .ToArray());

        var problemDetails =
            new ProblemDetails
            {
                Status = StatusCodes.Status400BadRequest,
                Title = "Doğrulama hatası.",
                Detail =
                    "Gönderilen bilgilerden biri veya birkaçı geçersizdir."
            };

        problemDetails.Extensions["errors"] = errors;

        return problemDetails;
    }
}

RefreshTokenCommandHandler
Burası token rotation'ın ana noktası.
using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Authentication.Responses;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Interfaces.Authorization;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Domain.Entities.Identity;

namespace MSKBS.Application.Authentication.Commands.RefreshToken;

/// <summary>
/// Refresh token'ı doğrular, eski token'ı revoke eder ve rotation
/// yöntemiyle yeni access/refresh token çifti oluşturur.
/// </summary>
public sealed class RefreshTokenCommandHandler
    : IRequestHandler<RefreshTokenCommand, RefreshResult>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly IJwtTokenService _jwtTokenService;
    private readonly IUserAuthorizationService _authorizationService;
    private readonly ICurrentUserService _currentUserService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// RefreshTokenCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public RefreshTokenCommandHandler(
        IMSKBSDbContext dbContext,
        IJwtTokenService jwtTokenService,
        IUserAuthorizationService authorizationService,
        ICurrentUserService currentUserService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _jwtTokenService = jwtTokenService;
        _authorizationService = authorizationService;
        _currentUserService = currentUserService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task<RefreshResult> Handle(
        RefreshTokenCommand request,
        CancellationToken cancellationToken)
    {
        var utcNow =
            _timeProvider.GetUtcNow().UtcDateTime;

        var tokenHash =
            _jwtTokenService.HashRefreshToken(
                request.RefreshToken);

        /*
         * Revoke edilmiş token'ları da görebilmemiz gerekir.
         * Token reuse saldırısını tespit edebilmek için global
         * query filter'ları bilinçli olarak kapatıyoruz.
         */
        var storedToken = await _dbContext.RefreshTokens
            .IgnoreQueryFilters()
            .Include(token => token.User)
            .FirstOrDefaultAsync(
                token => token.TokenHash == tokenHash,
                cancellationToken);

        if (storedToken is null)
        {
            throw new InvalidRefreshTokenException();
        }

        if (storedToken.RevokedAt.HasValue)
        {
            await RevokeReplacementChainAsync(
                storedToken,
                utcNow,
                cancellationToken);

            throw new InvalidRefreshTokenException();
        }

        if (storedToken.IsDeleted ||
            !storedToken.IsActive ||
            storedToken.ExpiresAt <= utcNow)
        {
            throw new InvalidRefreshTokenException();
        }

        var user = storedToken.User;

        if (user.IsDeleted ||
            !user.IsActive)
        {
            throw new InvalidRefreshTokenException();
        }

        if (user.IsLocked)
        {
            if (!user.LockoutEnd.HasValue ||
                user.LockoutEnd.Value > utcNow)
            {
                throw new AccountLockedException(
                    user.LockoutEnd);
            }

            user.IsLocked = false;
            user.LockoutEnd = null;
            user.AccessFailedCount = 0;
        }

        if (user.ValidFrom.HasValue &&
            user.ValidFrom.Value > utcNow)
        {
            throw new InvalidRefreshTokenException();
        }

        if (user.ValidTo.HasValue &&
            user.ValidTo.Value <= utcNow)
        {
            throw new InvalidRefreshTokenException();
        }

        /*
         * Refresh sırasında authorization bilgilerini DB'den yeniden
         * hesaplıyoruz. Böylece login sonrasında rol veya permission
         * değişmişse yeni access token güncel yetkilerle oluşturulur.
         */
        var authorization =
            await _authorizationService.GetAsync(
                user.Id,
                cancellationToken);

        var accessToken =
            _jwtTokenService.GenerateAccessToken(
                user,
                authorization.Roles,
                authorization.Permissions);

        var newRefreshToken =
            _jwtTokenService.GenerateRefreshToken();

        var newRefreshTokenEntity =
            new MSKBS.Domain.Entities.Identity.RefreshToken
            {
                UserId = user.Id,
                TokenHash =
                    newRefreshToken.TokenHash,

                ExpiresAt =
                    newRefreshToken.ExpiresAt,

                CreatedByIp =
                    _currentUserService.IpAddress,

                UserAgent =
                    _currentUserService.UserAgent,

                /*
                 * Otomatik refresh işleminde cihaz adı eski token'dan
                 * taşınır.
                 */
                DeviceName =
                    storedToken.DeviceName
            };

        await _dbContext.RefreshTokens.AddAsync(
            newRefreshTokenEntity,
            cancellationToken);

        /*
         * Eski token rotation nedeniyle kullanılamaz hale getirilir.
         */
        storedToken.RevokedAt = utcNow;
        storedToken.RevokedByIp =
            _currentUserService.IpAddress;

        storedToken.RevocationReason =
            "Refresh token rotated.";

        storedToken.ReplacedByTokenId =
            newRefreshTokenEntity.Id;

        storedToken.IsActive = false;

        await _dbContext.SaveChangesAsync(
            cancellationToken);

        return new RefreshResult
        {
            RefreshToken =
                newRefreshToken.Token,

            RefreshTokenExpiresAt =
                newRefreshToken.ExpiresAt,

            Response = new RefreshResponse
            {
                AccessToken =
                    accessToken.Token,

                AccessTokenExpiresAt =
                    accessToken.ExpiresAt,

                User = new LoginUserResponse
                {
                    Id = user.Id,
                    UserName = user.UserName,
                    Email = user.Email,
                    FirstName = user.FirstName,
                    LastName = user.LastName,

                    OrganizationUnitId =
                        user.OrganizationUnitId,

                    Roles =
                        authorization.Roles,

                    Permissions =
                        authorization.Permissions
                }
            }
        };
    }

    /// <summary>
    /// Daha önce revoke edilmiş bir refresh token'ın tekrar kullanılması
    /// durumunda token'ın replacement zincirindeki sonraki token'ları
    /// güvenlik amacıyla revoke eder.
    /// </summary>
    private async Task RevokeReplacementChainAsync(
        MSKBS.Domain.Entities.Identity.RefreshToken token,
        DateTime utcNow,
        CancellationToken cancellationToken)
    {
        var nextTokenId =
            token.ReplacedByTokenId;

        var visitedIds =
            new HashSet<Guid>();

        while (nextTokenId.HasValue &&
               visitedIds.Add(nextTokenId.Value))
        {
            var replacementToken =
                await _dbContext.RefreshTokens
                    .IgnoreQueryFilters()
                    .FirstOrDefaultAsync(
                        item =>
                            item.Id == nextTokenId.Value,
                        cancellationToken);

            if (replacementToken is null)
            {
                break;
            }

            if (!replacementToken.RevokedAt.HasValue)
            {
                replacementToken.RevokedAt = utcNow;
                replacementToken.RevokedByIp =
                    _currentUserService.IpAddress;

                replacementToken.RevocationReason =
                    "Refresh token reuse detected.";

                replacementToken.IsActive = false;
            }

            nextTokenId =
                replacementToken.ReplacedByTokenId;
        }

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }
}

public sealed class LogoutCommandHandler
    : IRequestHandler<LogoutCommand>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly IJwtTokenService _jwtTokenService;
    private readonly ICurrentUserService _currentUserService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// LogoutCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public LogoutCommandHandler(
        IMSKBSDbContext dbContext,
        IJwtTokenService jwtTokenService,
        ICurrentUserService currentUserService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _jwtTokenService = jwtTokenService;
        _currentUserService = currentUserService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task Handle(
        LogoutCommand request,
        CancellationToken cancellationToken)
    {
        if (string.IsNullOrWhiteSpace(
                request.RefreshToken))
        {
            return;
        }

        var tokenHash =
            _jwtTokenService.HashRefreshToken(
                request.RefreshToken);

        var refreshToken =
            await _dbContext.RefreshTokens
                .IgnoreQueryFilters()
                .FirstOrDefaultAsync(
                    token =>
                        token.TokenHash == tokenHash,
                    cancellationToken);

        /*
         * Logout idempotent olmalıdır.
         * Token bulunamadıysa veya zaten revoke edilmişse hata vermiyoruz.
         */
        if (refreshToken is null ||
            refreshToken.RevokedAt.HasValue)
        {
            return;
        }

        refreshToken.RevokedAt =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime;

        refreshToken.RevokedByIp =
            _currentUserService.IpAddress;

        refreshToken.RevocationReason =
            "User logout.";

        refreshToken.IsActive = false;

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }
}

using MediatR;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using MSKBS.Application.Authentication.Commands.Login;
using MSKBS.Application.Authentication.Commands.Logout;
using MSKBS.Application.Authentication.Commands.RefreshToken;
using MSKBS.Application.Authentication.Responses;

namespace MSKBS.Api.Controllers;

/// <summary>
/// Login, token yenileme ve logout authentication
/// işlemlerini sağlayan API controller'ıdır.
/// </summary>
[ApiController]
[Route("api/auth")]
public sealed class AuthController : ControllerBase
{
    private const string RefreshTokenCookieName =
        "mskbs_refresh_token";

    private readonly ISender _sender;

    /// <summary>
    /// AuthController sınıfının yeni örneğini oluşturur.
    /// </summary>
    public AuthController(ISender sender)
    {
        _sender = sender;
    }

    /// <summary>
    /// Kullanıcı adı/e-posta ve parola ile kullanıcı girişini gerçekleştirir.
    /// </summary>
    [HttpPost("login")]
    [AllowAnonymous]
    [ProducesResponseType(
        typeof(LoginResponse),
        StatusCodes.Status200OK)]
    public async Task<ActionResult<LoginResponse>> Login(
        [FromBody] LoginCommand command,
        CancellationToken cancellationToken)
    {
        var result =
            await _sender.Send(
                command,
                cancellationToken);

        AppendRefreshTokenCookie(
            result.RefreshToken,
            result.RefreshTokenExpiresAt);

        return Ok(result.Response);
    }

    /// <summary>
    /// HttpOnly cookie içerisindeki refresh token kullanılarak
    /// yeni access ve refresh token üretir.
    /// </summary>
    [HttpPost("refresh")]
    [AllowAnonymous]
    [ProducesResponseType(
        typeof(RefreshResponse),
        StatusCodes.Status200OK)]
    public async Task<ActionResult<RefreshResponse>> Refresh(
        CancellationToken cancellationToken)
    {
        if (!Request.Cookies.TryGetValue(
                RefreshTokenCookieName,
                out var refreshToken) ||
            string.IsNullOrWhiteSpace(refreshToken))
        {
            return Unauthorized();
        }

        var result =
            await _sender.Send(
                new RefreshTokenCommand
                {
                    RefreshToken = refreshToken
                },
                cancellationToken);

        AppendRefreshTokenCookie(
            result.RefreshToken,
            result.RefreshTokenExpiresAt);

        return Ok(result.Response);
    }

    /// <summary>
    /// Mevcut refresh token oturumunu revoke eder ve refresh
    /// token cookie'sini istemciden kaldırır.
    /// </summary>
    [HttpPost("logout")]
    [AllowAnonymous]
    [ProducesResponseType(
        StatusCodes.Status204NoContent)]
    public async Task<IActionResult> Logout(
        CancellationToken cancellationToken)
    {
        if (Request.Cookies.TryGetValue(
                RefreshTokenCookieName,
                out var refreshToken) &&
            !string.IsNullOrWhiteSpace(refreshToken))
        {
            await _sender.Send(
                new LogoutCommand
                {
                    RefreshToken = refreshToken
                },
                cancellationToken);
        }

        DeleteRefreshTokenCookie();

        return NoContent();
    }

    /// <summary>
    /// Açık refresh token değerini güvenli HttpOnly cookie olarak
    /// response'a ekler.
    /// </summary>
    private void AppendRefreshTokenCookie(
        string refreshToken,
        DateTime expiresAt)
    {
        Response.Cookies.Append(
            RefreshTokenCookieName,
            refreshToken,
            new CookieOptions
            {
                HttpOnly = true,
                Secure = true,
                SameSite = SameSiteMode.Lax,

                Expires =
                    new DateTimeOffset(
                        DateTime.SpecifyKind(
                            expiresAt,
                            DateTimeKind.Utc)),

                IsEssential = true,
                Path = "/api/auth"
            });
    }

    /// <summary>
    /// Refresh token cookie'sini istemciden kaldırır.
    /// </summary>
    private void DeleteRefreshTokenCookie()
    {
        Response.Cookies.Delete(
            RefreshTokenCookieName,
            new CookieOptions
            {
                HttpOnly = true,
                Secure = true,
                SameSite = SameSiteMode.Lax,
                Path = "/api/auth"
            });
    }
}

using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Interfaces.Authorization;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Domain.Constants;

namespace MSKBS.Application.Common.Authorization;

/// <summary>
/// Mevcut kullanıcının permission durumunu güncel veritabanı kayıtları
/// üzerinden hesaplayan merkezi authorization servisidir.
/// </summary>
public sealed class PermissionChecker : IPermissionChecker
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly ICurrentUserService _currentUserService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// PermissionChecker sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="dbContext">
    /// Identity ve authorization kayıtlarına erişim sağlayan DbContext'tir.
    /// </param>
    /// <param name="currentUserService">
    /// Mevcut kullanıcının kimlik bilgilerini sağlar.
    /// </param>
    /// <param name="timeProvider">
    /// Hesap geçerlilik kontrollerinde kullanılacak UTC zamanı sağlar.
    /// </param>
    public PermissionChecker(
        IMSKBSDbContext dbContext,
        ICurrentUserService currentUserService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _currentUserService = currentUserService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task<bool> HasPermissionAsync(
        string permissionCode,
        CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(permissionCode))
        {
            return false;
        }

        var userId = _currentUserService.UserId;

        if (!userId.HasValue)
        {
            return false;
        }

        var utcNow =
            _timeProvider.GetUtcNow().UtcDateTime;

        /*
         * Global query filter sayesinde yalnızca aktif ve
         * soft delete edilmemiş kullanıcılar bulunur.
         *
         * Ayrıca hesabın tarihsel geçerliliği ve kilit durumu da
         * authorization sırasında tekrar kontrol edilir.
         */
        var userState = await _dbContext.Users
            .Where(user => user.Id == userId.Value)
            .Select(user => new
            {
                user.IsLocked,
                user.LockoutEnd,
                user.ValidFrom,
                user.ValidTo
            })
            .SingleOrDefaultAsync(cancellationToken);

        if (userState is null)
        {
            return false;
        }

        if (userState.IsLocked &&
            (!userState.LockoutEnd.HasValue ||
             userState.LockoutEnd.Value > utcNow))
        {
            return false;
        }

        if (userState.ValidFrom.HasValue &&
            userState.ValidFrom.Value > utcNow)
        {
            return false;
        }

        if (userState.ValidTo.HasValue &&
            userState.ValidTo.Value <= utcNow)
        {
            return false;
        }

        /*
         * 1. ADMIN kontrolü.
         *
         * JWT claim'ine güvenmek yerine güncel DB kaydı kontrol edilir.
         * Böylece ADMIN rolü kaldırılırsa eski access token tam yetki
         * vermeye devam etmez.
         */
        var isAdmin = await _dbContext.UserRoles
            .AnyAsync(
                userRole =>
                    userRole.UserId == userId.Value &&
                    userRole.Role.Code == SystemRoles.Admin,
                cancellationToken);

        if (isAdmin)
        {
            return true;
        }

        /*
         * 2. Kullanıcı bazlı override kontrolü.
         *
         * nullable bool kullanmamızın nedeni:
         * null  = override yok
         * true  = açıkça izin verilmiş
         * false = açıkça reddedilmiş
         */
        var userOverride = await _dbContext.UserPermissions
            .Where(userPermission =>
                userPermission.UserId == userId.Value &&
                userPermission.Permission.Code == permissionCode)
            .Select(userPermission =>
                (bool?)userPermission.IsGranted)
            .SingleOrDefaultAsync(cancellationToken);

        if (userOverride.HasValue)
        {
            return userOverride.Value;
        }

        /*
         * 3. Kullanıcının rollerinden herhangi biri permission'a
         * sahipse erişim verilir.
         */
        var hasRolePermission =
            await _dbContext.RolePermissions
                .AnyAsync(
                    rolePermission =>
                        rolePermission.Permission.Code == permissionCode &&
                        _dbContext.UserRoles.Any(
                            userRole =>
                                userRole.UserId == userId.Value &&
                                userRole.RoleId == rolePermission.RoleId),
                    cancellationToken);

        return hasRolePermission;
    }
}

5. PermissionAuthorizationHandler
PermissionAuthorizationHandler.cs
using Microsoft.AspNetCore.Authorization;
using MSKBS.Application.Common.Interfaces.Authorization;

namespace MSKBS.Infrastructure.Authorization;

/// <summary>
/// PermissionRequirement authorization taleplerini IPermissionChecker
/// üzerinden güncel veritabanı bilgileriyle doğrular.
/// </summary>
public sealed class PermissionAuthorizationHandler
    : AuthorizationHandler<PermissionRequirement>
{
    private readonly IPermissionChecker _permissionChecker;

    /// <summary>
    /// PermissionAuthorizationHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="permissionChecker">
    /// Kullanıcının gerçek permission durumunu kontrol eden servistir.
    /// </param>
    public PermissionAuthorizationHandler(
        IPermissionChecker permissionChecker)
    {
        _permissionChecker = permissionChecker;
    }

    /// <summary>
    /// Kullanıcının requirement içerisinde belirtilen permission'a
    /// sahip olup olmadığını kontrol eder.
    /// </summary>
    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        PermissionRequirement requirement)
    {
        /*
         * Kullanıcı authentication olmamışsa permission sorgusu
         * çalıştırmaya gerek yoktur.
         */
        if (context.User.Identity?.IsAuthenticated != true)
        {
            return;
        }

        var hasPermission =
            await _permissionChecker.HasPermissionAsync(
                requirement.PermissionCode);

        if (hasPermission)
        {
            context.Succeed(requirement);
        }
    }
}

PermissionAuthorizationPolicyProvider.cs
using Microsoft.AspNetCore.Authorization;
using Microsoft.Extensions.Options;
using MSKBS.Application.Common.Constants;

namespace MSKBS.Infrastructure.Authorization;

/// <summary>
/// "Permission:" prefix'i ile başlayan authorization policy'lerini
/// çalışma zamanında dinamik olarak oluşturan policy provider'dır.
/// </summary>
public sealed class PermissionAuthorizationPolicyProvider
    : DefaultAuthorizationPolicyProvider
{
    /// <summary>
    /// PermissionAuthorizationPolicyProvider sınıfının yeni örneğini oluşturur.
    /// </summary>
    /// <param name="options">
    /// Uygulamanın authorization ayarlarını içerir.
    /// </param>
    public PermissionAuthorizationPolicyProvider(
        IOptions<AuthorizationOptions> options)
        : base(options)
    {
    }

    /// <summary>
    /// İstenen policy permission policy'si ise çalışma zamanında oluşturur;
    /// diğer policy'ler için varsayılan ASP.NET Core provider'ına yönlendirir.
    /// </summary>
    /// <param name="policyName">
    /// Oluşturulması istenen authorization policy adıdır.
    /// </param>
    /// <returns>
    /// Bulunan veya dinamik olarak oluşturulan authorization policy'sini döndürür.
    /// </returns>
    public override Task<AuthorizationPolicy?> GetPolicyAsync(
        string policyName)
    {
        if (!policyName.StartsWith(
                AuthorizationPolicyNames.PermissionPrefix,
                StringComparison.OrdinalIgnoreCase))
        {
            return base.GetPolicyAsync(policyName);
        }

        var permissionCode =
            policyName[
                AuthorizationPolicyNames
                    .PermissionPrefix.Length..];

        if (string.IsNullOrWhiteSpace(permissionCode))
        {
            return Task.FromResult<AuthorizationPolicy?>(
                null);
        }

        var policy =
            new AuthorizationPolicyBuilder()
                .RequireAuthenticatedUser()
                .AddRequirements(
                    new PermissionRequirement(
                        permissionCode))
                .Build();

        return Task.FromResult<AuthorizationPolicy?>(
            policy);
    }
}

using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Constants;

AddJwtBearer:
.AddJwtBearer(options =>
{
    options.MapInboundClaims = true;

    options.TokenValidationParameters =
        new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidIssuer = jwtSettings.Issuer,

            ValidateAudience = true,
            ValidAudience = jwtSettings.Audience,

            ValidateLifetime = true,

            ValidateIssuerSigningKey = true,

            IssuerSigningKey =
                new SymmetricSecurityKey(
                    Encoding.UTF8.GetBytes(
                        jwtSettings.SecretKey)),

            ClockSkew =
                TimeSpan.FromSeconds(30),

            NameClaimType =
                ClaimTypes.Name,

            RoleClaimType =
                ClaimTypes.Role
        };

    options.Events =
        new JwtBearerEvents
        {
            OnTokenValidated = async context =>
            {
                var userIdValue =
                    context.Principal?
                        .FindFirstValue(
                            ClaimTypes.NameIdentifier);

                var tokenSecurityStamp =
                    context.Principal?
                        .FindFirstValue(
                            CustomClaimTypes.SecurityStamp);

                if (!Guid.TryParse(
                        userIdValue,
                        out var userId) ||
                    string.IsNullOrWhiteSpace(
                        tokenSecurityStamp))
                {
                    context.Fail(
                        "Geçersiz kullanıcı kimliği.");

                    return;
                }

                var dbContext =
                    context.HttpContext
                        .RequestServices
                        .GetRequiredService<IMSKBSDbContext>();

                var utcNow =
                    TimeProvider.System
                        .GetUtcNow()
                        .UtcDateTime;

                var user = await dbContext.Users
                    .AsNoTracking()
                    .Where(x => x.Id == userId)
                    .Select(x => new
                    {
                        x.SecurityStamp,
                        x.IsLocked,
                        x.LockoutEnd,
                        x.ValidFrom,
                        x.ValidTo
                    })
                    .SingleOrDefaultAsync(
                        context.HttpContext
                            .RequestAborted);

                /*
                 * Global query filter sayesinde kullanıcı soft delete edilmiş
                 * veya pasif hale getirilmişse burada zaten bulunamaz.
                 */
                if (user is null)
                {
                    context.Fail(
                        "Kullanıcı artık aktif değildir.");

                    return;
                }

                if (!string.Equals(
                        user.SecurityStamp,
                        tokenSecurityStamp,
                        StringComparison.Ordinal))
                {
                    context.Fail(
                        "Oturum artık geçerli değildir.");

                    return;
                }

                if (user.IsLocked &&
                    (!user.LockoutEnd.HasValue ||
                     user.LockoutEnd.Value > utcNow))
                {
                    context.Fail(
                        "Kullanıcı hesabı kilitlidir.");

                    return;
                }

                if (user.ValidFrom.HasValue &&
                    user.ValidFrom.Value > utcNow)
                {
                    context.Fail(
                        "Kullanıcı hesabı henüz geçerli değildir.");

                    return;
                }

                if (user.ValidTo.HasValue &&
                    user.ValidTo.Value <= utcNow)
                {
                    context.Fail(
                        "Kullanıcı hesabının geçerlilik süresi dolmuştur.");
                }
            }
        };
});

GetCurrentUserQueryHandler.cs
using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Authentication.Responses;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Authorization;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;

namespace MSKBS.Application.Authentication.Queries.GetCurrentUser;

/// <summary>
/// Mevcut kullanıcının profil, organizasyon, rol ve effective permission
/// bilgilerini güncel veritabanı kayıtlarından yükler.
/// </summary>
public sealed class GetCurrentUserQueryHandler
    : IRequestHandler<GetCurrentUserQuery, CurrentUserResponse>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly ICurrentUserService _currentUserService;
    private readonly IUserAuthorizationService _authorizationService;

    /// <summary>
    /// GetCurrentUserQueryHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public GetCurrentUserQueryHandler(
        IMSKBSDbContext dbContext,
        ICurrentUserService currentUserService,
        IUserAuthorizationService authorizationService)
    {
        _dbContext = dbContext;
        _currentUserService = currentUserService;
        _authorizationService = authorizationService;
    }

    /// <inheritdoc />
    public async Task<CurrentUserResponse> Handle(
        GetCurrentUserQuery request,
        CancellationToken cancellationToken)
    {
        var userId =
            _currentUserService.UserId
            ?? throw new AuthenticationRequiredException();

        var user = await _dbContext.Users
            .AsNoTracking()
            .Where(x => x.Id == userId)
            .Select(x => new
            {
                x.Id,
                x.UserName,
                x.Email,
                x.FirstName,
                x.LastName,
                x.PhoneNumber,
                x.OrganizationUnitId,

                OrganizationUnitName =
                    x.OrganizationUnit != null
                        ? x.OrganizationUnit.Name
                        : null,

                x.LastLoginDate,
                x.PasswordChangedDate
            })
            .SingleOrDefaultAsync(
                cancellationToken)
            ?? throw new AuthenticationRequiredException();

        var authorization =
            await _authorizationService.GetAsync(
                user.Id,
                cancellationToken);

        return new CurrentUserResponse
        {
            Id = user.Id,
            UserName = user.UserName,
            Email = user.Email,
            FirstName = user.FirstName,
            LastName = user.LastName,

            FullName =
                $"{user.FirstName} {user.LastName}".Trim(),

            PhoneNumber = user.PhoneNumber,

            OrganizationUnitId =
                user.OrganizationUnitId,

            OrganizationUnitName =
                user.OrganizationUnitName,

            LastLoginDate =
                user.LastLoginDate,

            PasswordChangedDate =
                user.PasswordChangedDate,

            Roles =
                authorization.Roles,

            Permissions =
                authorization.Permissions
        };
    }
}

ChangePasswordCommandValidator.cs
using FluentValidation;

namespace MSKBS.Application.Authentication.Commands.ChangePassword;

/// <summary>
/// ChangePasswordCommand için parola değiştirme validation
/// kurallarını tanımlar.
/// </summary>
public sealed class ChangePasswordCommandValidator
    : AbstractValidator<ChangePasswordCommand>
{
    /// <summary>
    /// Parola değiştirme doğrulama kurallarını oluşturur.
    /// </summary>
    public ChangePasswordCommandValidator()
    {
        RuleFor(x => x.CurrentPassword)
            .NotEmpty()
            .MaximumLength(256);

        RuleFor(x => x.NewPassword)
            .NotEmpty()
            .MinimumLength(8)
            .MaximumLength(128)
            .Matches("[A-Z]")
            .WithMessage(
                "Yeni parola en az bir büyük harf içermelidir.")
            .Matches("[a-z]")
            .WithMessage(
                "Yeni parola en az bir küçük harf içermelidir.")
            .Matches("[0-9]")
            .WithMessage(
                "Yeni parola en az bir rakam içermelidir.");

        RuleFor(x => x.ConfirmPassword)
            .Equal(x => x.NewPassword)
            .WithMessage(
                "Yeni parola ve parola tekrarı eşleşmiyor.");

        RuleFor(x => x.NewPassword)
            .NotEqual(x => x.CurrentPassword)
            .WithMessage(
                "Yeni parola mevcut parola ile aynı olamaz.");
    }
}

ChangePasswordCommandHandler.cs
using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;

namespace MSKBS.Application.Authentication.Commands.ChangePassword;

/// <summary>
/// Mevcut kullanıcı parolasını doğrular, yeni parolayı hash'leyerek
/// kaydeder ve kullanıcının bütün aktif refresh token oturumlarını sonlandırır.
/// </summary>
public sealed class ChangePasswordCommandHandler
    : IRequestHandler<ChangePasswordCommand>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly ICurrentUserService _currentUserService;
    private readonly IPasswordHasherService _passwordHasherService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// ChangePasswordCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public ChangePasswordCommandHandler(
        IMSKBSDbContext dbContext,
        ICurrentUserService currentUserService,
        IPasswordHasherService passwordHasherService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _currentUserService = currentUserService;
        _passwordHasherService = passwordHasherService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task Handle(
        ChangePasswordCommand request,
        CancellationToken cancellationToken)
    {
        var userId =
            _currentUserService.UserId
            ?? throw new AuthenticationRequiredException();

        var user = await _dbContext.Users
            .SingleOrDefaultAsync(
                x => x.Id == userId,
                cancellationToken)
            ?? throw new AuthenticationRequiredException();

        var currentPasswordValid =
            _passwordHasherService.VerifyPassword(
                user.PasswordHash,
                request.CurrentPassword);

        if (!currentPasswordValid)
        {
            throw new InvalidCurrentPasswordException();
        }

        var utcNow =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime;

        user.PasswordHash =
            _passwordHasherService.HashPassword(
                request.NewPassword);

        user.PasswordChangedDate = utcNow;

        /*
         * Yeni security stamp üretilmesi kullanıcının eski access
         * token'larının JWT validation aşamasında geçersiz olmasını sağlar.
         */
        user.SecurityStamp =
            Guid.NewGuid().ToString("N");

        await RevokeAllRefreshTokensAsync(
            userId,
            utcNow,
            "Password changed.",
            cancellationToken);

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }

    /// <summary>
    /// Kullanıcının revoke edilmemiş bütün refresh token oturumlarını
    /// güvenlik amacıyla sonlandırır.
    /// </summary>
    private async Task RevokeAllRefreshTokensAsync(
        Guid userId,
        DateTime utcNow,
        string reason,
        CancellationToken cancellationToken)
    {
        var refreshTokens =
            await _dbContext.RefreshTokens
                .IgnoreQueryFilters()
                .Where(token =>
                    token.UserId == userId &&
                    !token.RevokedAt.HasValue)
                .ToListAsync(
                    cancellationToken);

        foreach (var token in refreshTokens)
        {
            token.RevokedAt = utcNow;

            token.RevokedByIp =
                _currentUserService.IpAddress;

            token.RevocationReason = reason;
            token.IsActive = false;
        }
    }
}

using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;

namespace MSKBS.Application.Authentication.Commands.LogoutAll;

/// <summary>
/// Kullanıcının bütün refresh token kayıtlarını revoke eder ve
/// SecurityStamp değerini değiştirerek bütün mevcut access token'ları
/// geçersiz hale getirir.
/// </summary>
public sealed class LogoutAllCommandHandler
    : IRequestHandler<LogoutAllCommand>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly ICurrentUserService _currentUserService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// LogoutAllCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public LogoutAllCommandHandler(
        IMSKBSDbContext dbContext,
        ICurrentUserService currentUserService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _currentUserService = currentUserService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task Handle(
        LogoutAllCommand request,
        CancellationToken cancellationToken)
    {
        var userId =
            _currentUserService.UserId
            ?? throw new AuthenticationRequiredException();

        var user = await _dbContext.Users
            .SingleOrDefaultAsync(
                x => x.Id == userId,
                cancellationToken)
            ?? throw new AuthenticationRequiredException();

        var utcNow =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime;

        /*
         * SecurityStamp değiştirilerek mevcut bütün access token'lar
         * bir sonraki API çağrısında geçersiz hale getirilir.
         */
        user.SecurityStamp =
            Guid.NewGuid().ToString("N");

        var refreshTokens =
            await _dbContext.RefreshTokens
                .IgnoreQueryFilters()
                .Where(token =>
                    token.UserId == userId &&
                    !token.RevokedAt.HasValue)
                .ToListAsync(
                    cancellationToken);

        foreach (var token in refreshTokens)
        {
            token.RevokedAt = utcNow;

            token.RevokedByIp =
                _currentUserService.IpAddress;

            token.RevocationReason =
                "Logout from all devices.";

            token.IsActive = false;
        }

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }
}

4. PasswordResetTokenService
Infrastructure/Authentication/PasswordResetTokenService.cs
using System.Security.Cryptography;
using System.Text;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.Tokens;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Models.Authentication;
using MSKBS.Application.Common.Options;

namespace MSKBS.Infrastructure.Authentication;

/// <summary>
/// Parola sıfırlama işlemleri için kriptografik olarak güvenli
/// rastgele token üretir ve token hash işlemlerini gerçekleştirir.
/// </summary>
public sealed class PasswordResetTokenService
    : IPasswordResetTokenService
{
    private const int TokenByteLength = 64;

    private readonly PasswordResetSettings _settings;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// PasswordResetTokenService sınıfının yeni örneğini oluşturur.
    /// </summary>
    public PasswordResetTokenService(
        IOptions<PasswordResetSettings> options,
        TimeProvider timeProvider)
    {
        _settings = options.Value;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public PasswordResetTokenResult GenerateToken()
    {
        var randomBytes =
            RandomNumberGenerator.GetBytes(TokenByteLength);

        var token =
            Base64UrlEncoder.Encode(randomBytes);

        var expiresAt =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime
                .AddMinutes(
                    _settings.TokenExpirationMinutes);

        return new PasswordResetTokenResult
        {
            Token = token,
            TokenHash = HashToken(token),
            ExpiresAt = expiresAt
        };
    }

    /// <inheritdoc />
    public string HashToken(string token)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(token);

        var bytes =
            Encoding.UTF8.GetBytes(token);

        var hash =
            SHA256.HashData(bytes);

        return Convert.ToHexString(hash);
    }
}

using MailKit.Net.Smtp;
using MailKit.Security;
using Microsoft.Extensions.Options;
using MimeKit;
using MSKBS.Application.Common.Interfaces.Communication;

namespace MSKBS.Infrastructure.Communication;

/// <summary>
/// MailKit kullanarak SMTP üzerinden e-posta gönderen servis
/// implementasyonudur.
/// </summary>
public sealed class SmtpEmailSender : IEmailSender
{
    private readonly SmtpSettings _settings;

    /// <summary>
    /// SmtpEmailSender sınıfının yeni örneğini oluşturur.
    /// </summary>
    public SmtpEmailSender(
        IOptions<SmtpSettings> options)
    {
        _settings = options.Value;
    }

    /// <inheritdoc />
    public async Task SendAsync(
        string to,
        string subject,
        string htmlBody,
        CancellationToken cancellationToken = default)
    {
        var message = new MimeMessage();

        message.From.Add(
            new MailboxAddress(
                _settings.FromName,
                _settings.FromAddress));

        message.To.Add(
            MailboxAddress.Parse(to));

        message.Subject = subject;

        message.Body =
            new TextPart("html")
            {
                Text = htmlBody
            };

        using var client = new SmtpClient();

        await client.ConnectAsync(
            _settings.Host,
            _settings.Port,
            SecureSocketOptions.Auto,
            cancellationToken);

        if (!string.IsNullOrWhiteSpace(
                _settings.UserName))
        {
            await client.AuthenticateAsync(
                _settings.UserName,
                _settings.Password,
                cancellationToken);
        }

        await client.SendAsync(
            message,
            cancellationToken);

        await client.DisconnectAsync(
            true,
            cancellationToken);
    }
}

using System.Net;
using MediatR;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Interfaces.Communication;
using MSKBS.Application.Common.Interfaces.Persistence;
using MSKBS.Application.Common.Options;
using MSKBS.Domain.Entities.Identity;

namespace MSKBS.Application.Authentication.Commands.ForgotPassword;

/// <summary>
/// Kullanıcı için güvenli parola sıfırlama token'ı oluşturur,
/// önceki aktif token'ları geçersiz hale getirir ve reset bağlantısını
/// kullanıcının e-posta adresine gönderir.
/// </summary>
public sealed class ForgotPasswordCommandHandler
    : IRequestHandler<ForgotPasswordCommand>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly IPasswordResetTokenService _tokenService;
    private readonly IEmailSender _emailSender;
    private readonly PasswordResetSettings _settings;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// ForgotPasswordCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public ForgotPasswordCommandHandler(
        IMSKBSDbContext dbContext,
        IPasswordResetTokenService tokenService,
        IEmailSender emailSender,
        IOptions<PasswordResetSettings> settings,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _tokenService = tokenService;
        _emailSender = emailSender;
        _settings = settings.Value;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task Handle(
        ForgotPasswordCommand request,
        CancellationToken cancellationToken)
    {
        var normalizedEmail =
            request.Email
                .Trim()
                .ToUpperInvariant();

        /*
         * Global filter nedeniyle yalnızca aktif ve silinmemiş
         * kullanıcılar parola sıfırlama talebi oluşturabilir.
         */
        var user = await _dbContext.Users
            .FirstOrDefaultAsync(
                x => x.NormalizedEmail == normalizedEmail,
                cancellationToken);

        /*
         * Güvenlik nedeniyle kullanıcı bulunamadığında hata dönmüyoruz.
         * Endpoint her durumda aynı response'u üretir.
         */
        if (user is null)
        {
            return;
        }

        var utcNow =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime;

        /*
         * Kullanıcının daha önce oluşturulmuş ve henüz kullanılmamış
         * token'larını geçersiz hale getiriyoruz.
         */
        var previousTokens =
            await _dbContext.PasswordResetTokens
                .IgnoreQueryFilters()
                .Where(token =>
                    token.UserId == user.Id &&
                    !token.UsedAt.HasValue &&
                    !token.RevokedAt.HasValue)
                .ToListAsync(
                    cancellationToken);

        foreach (var token in previousTokens)
        {
            token.RevokedAt = utcNow;
            token.IsActive = false;
        }

        var generatedToken =
            _tokenService.GenerateToken();

        await _dbContext.PasswordResetTokens.AddAsync(
            new PasswordResetToken
            {
                UserId = user.Id,
                TokenHash =
                    generatedToken.TokenHash,

                ExpiresAt =
                    generatedToken.ExpiresAt
            },
            cancellationToken);

        await _dbContext.SaveChangesAsync(
            cancellationToken);

        var resetUrl =
            BuildResetUrl(
                generatedToken.Token);

        var htmlBody =
            BuildEmailBody(
                user.FirstName,
                resetUrl,
                generatedToken.ExpiresAt);

        await _emailSender.SendAsync(
            user.Email,
            "MSKBS Parola Sıfırlama",
            htmlBody,
            cancellationToken);
    }

    /// <summary>
    /// Angular uygulamasındaki parola sıfırlama ekranına yönlendirecek
    /// güvenli reset URL'ini oluşturur.
    /// </summary>
    private string BuildResetUrl(
        string token)
    {
        var encodedToken =
            Uri.EscapeDataString(token);

        return
            $"{_settings.ResetUrlBase}?token={encodedToken}";
    }

    /// <summary>
    /// Kullanıcıya gönderilecek parola sıfırlama e-posta içeriğini oluşturur.
    /// </summary>
    private static string BuildEmailBody(
        string firstName,
        string resetUrl,
        DateTime expiresAt)
    {
        var safeName =
            WebUtility.HtmlEncode(firstName);

        var safeUrl =
            WebUtility.HtmlEncode(resetUrl);

        return $"""
                <p>Merhaba {safeName},</p>

                <p>MSKBS hesabınız için parola sıfırlama talebi alındı.</p>

                <p>
                    <a href="{safeUrl}">
                        Parolamı Sıfırla
                    </a>
                </p>

                <p>
                    Bu bağlantı {expiresAt:dd.MM.yyyy HH:mm} UTC tarihine
                    kadar geçerlidir.
                </p>

                <p>
                    Bu talebi siz oluşturmadıysanız herhangi bir işlem
                    yapmanız gerekmez.
                </p>
                """;
    }
}

13. ResetPasswordCommandValidator
using FluentValidation;

namespace MSKBS.Application.Authentication.Commands.ResetPassword;

/// <summary>
/// ResetPasswordCommand validation kurallarını tanımlar.
/// </summary>
public sealed class ResetPasswordCommandValidator
    : AbstractValidator<ResetPasswordCommand>
{
    /// <summary>
    /// Parola sıfırlama doğrulama kurallarını oluşturur.
    /// </summary>
    public ResetPasswordCommandValidator()
    {
        RuleFor(x => x.Token)
            .NotEmpty();

        RuleFor(x => x.NewPassword)
            .NotEmpty()
            .MinimumLength(8)
            .MaximumLength(128)
            .Matches("[A-Z]")
            .WithMessage(
                "Parola en az bir büyük harf içermelidir.")
            .Matches("[a-z]")
            .WithMessage(
                "Parola en az bir küçük harf içermelidir.")
            .Matches("[0-9]")
            .WithMessage(
                "Parola en az bir rakam içermelidir.");

        RuleFor(x => x.ConfirmPassword)
            .Equal(x => x.NewPassword)
            .WithMessage(
                "Parola ve parola tekrarı eşleşmiyor.");

15. ResetPasswordCommandHandler
using MediatR;
using Microsoft.EntityFrameworkCore;
using MSKBS.Application.Common.Exceptions;
using MSKBS.Application.Common.Interfaces.Authentication;
using MSKBS.Application.Common.Interfaces.Identity;
using MSKBS.Application.Common.Interfaces.Persistence;

namespace MSKBS.Application.Authentication.Commands.ResetPassword;

/// <summary>
/// Geçerli parola sıfırlama token'ını doğrular, kullanıcı parolasını
/// değiştirir ve bütün mevcut authentication oturumlarını sonlandırır.
/// </summary>
public sealed class ResetPasswordCommandHandler
    : IRequestHandler<ResetPasswordCommand>
{
    private readonly IMSKBSDbContext _dbContext;
    private readonly IPasswordResetTokenService _tokenService;
    private readonly IPasswordHasherService _passwordHasherService;
    private readonly TimeProvider _timeProvider;

    /// <summary>
    /// ResetPasswordCommandHandler sınıfının yeni örneğini oluşturur.
    /// </summary>
    public ResetPasswordCommandHandler(
        IMSKBSDbContext dbContext,
        IPasswordResetTokenService tokenService,
        IPasswordHasherService passwordHasherService,
        TimeProvider timeProvider)
    {
        _dbContext = dbContext;
        _tokenService = tokenService;
        _passwordHasherService = passwordHasherService;
        _timeProvider = timeProvider;
    }

    /// <inheritdoc />
    public async Task Handle(
        ResetPasswordCommand request,
        CancellationToken cancellationToken)
    {
        var utcNow =
            _timeProvider
                .GetUtcNow()
                .UtcDateTime;

        var tokenHash =
            _tokenService.HashToken(
                request.Token);

        var resetToken =
            await _dbContext.PasswordResetTokens
                .IgnoreQueryFilters()
                .Include(token => token.User)
                .SingleOrDefaultAsync(
                    token =>
                        token.TokenHash == tokenHash,
                    cancellationToken);

        if (resetToken is null ||
            resetToken.IsDeleted ||
            !resetToken.IsActive ||
            resetToken.UsedAt.HasValue ||
            resetToken.RevokedAt.HasValue ||
            resetToken.ExpiresAt <= utcNow)
        {
            throw new InvalidPasswordResetTokenException();
        }

        var user =
            resetToken.User;

        if (user.IsDeleted ||
            !user.IsActive)
        {
            throw new InvalidPasswordResetTokenException();
        }

        user.PasswordHash =
            _passwordHasherService.HashPassword(
                request.NewPassword);

        user.PasswordChangedDate = utcNow;

        /*
         * Eski access token'ların anında geçersiz hale gelmesini sağlar.
         */
        user.SecurityStamp =
            Guid.NewGuid().ToString("N");

        user.AccessFailedCount = 0;

        /*
         * Geçici lockout varsa parola sıfırlama sonrasında kaldırılır.
         * LockoutEnd null ise bu manuel/süresiz kilit olarak kabul edilir
         * ve parola sıfırlama işlemi kilidi kaldırmaz.
         */
        if (user.IsLocked &&
            user.LockoutEnd.HasValue)
        {
            user.IsLocked = false;
            user.LockoutEnd = null;
        }

        resetToken.UsedAt = utcNow;
        resetToken.IsActive = false;

        /*
         * Aynı kullanıcıya ait diğer kullanılmamış reset token'larını da
         * geçersiz hale getiriyoruz.
         */
        var otherResetTokens =
            await _dbContext.PasswordResetTokens
                .IgnoreQueryFilters()
                .Where(token =>
                    token.UserId == user.Id &&
                    token.Id != resetToken.Id &&
                    !token.UsedAt.HasValue &&
                    !token.RevokedAt.HasValue)
                .ToListAsync(
                    cancellationToken);

        foreach (var token in otherResetTokens)
        {
            token.RevokedAt = utcNow;
            token.IsActive = false;
        }

        /*
         * Parola değiştiği için bütün cihazlardaki refresh token
         * oturumları kapatılır.
         */
        var refreshTokens =
            await _dbContext.RefreshTokens
                .IgnoreQueryFilters()
                .Where(token =>
                    token.UserId == user.Id &&
                    !token.RevokedAt.HasValue)
                .ToListAsync(
                    cancellationToken);

        foreach (var refreshToken in refreshTokens)
        {
            refreshToken.RevokedAt = utcNow;
            refreshToken.RevocationReason =
                "Password reset.";

            refreshToken.IsActive = false;
        }

        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }
}

Böylece reset sonrası:
Eski AccessToken       ❌ SecurityStamp değişti
Eski RefreshToken      ❌ revoke edildi
Eski ResetToken        ❌ UsedAt
Diğer ResetToken'lar   ❌ revoke edildi
